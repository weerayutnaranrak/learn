model/product.go — Model (Data + Business Logic)Go
package model

import (
    "database/sql"
    "errors"
    "time"
)

// ── Model: ข้อมูลและ Business Rules ──────────────────────────────────────
// ไม่รู้จัก HTTP, ไม่รู้จัก HTML template

type Product struct {
    ID          int       `json:"id"`
    Name        string    `json:"name"`
    Description string    `json:"description"`
    Price       float64   `json:"price"`
    Stock       int       `json:"stock"`
    CreatedAt   time.Time `json:"created_at"`
}

// Business rule
func (p *Product) Validate() error {
    if p.Name == "" {
        return errors.New("name is required")
    }
    if p.Price < 0 {
        return errors.New("price cannot be negative")
    }
    if p.Stock < 0 {
        return errors.New("stock cannot be negative")
    }
    return nil
}

func (p *Product) IsAvailable() bool { return p.Stock > 0 }

// ProductModel — Data Access (อาจแยกเป็น Repository ก็ได้)
type ProductModel struct{ DB *sql.DB }

func (m *ProductModel) All() ([]*Product, error) {
    rows, err := m.DB.Query(
        `SELECT id,name,description,price,stock,created_at FROM products ORDER BY created_at DESC`,
    )
    if err != nil { return nil, err }
    defer rows.Close()
    var products []*Product
    for rows.Next() {
        var p Product
        rows.Scan(&p.ID, &p.Name, &p.Description, &p.Price, &p.Stock, &p.CreatedAt)
        products = append(products, &p)
    }
    return products, nil
}

func (m *ProductModel) ByID(id int) (*Product, error) {
    var p Product
    err := m.DB.QueryRow(
        `SELECT id,name,description,price,stock,created_at FROM products WHERE id=$1`, id,
    ).Scan(&p.ID, &p.Name, &p.Description, &p.Price, &p.Stock, &p.CreatedAt)
    if err == sql.ErrNoRows { return nil, errors.New("product not found") }
    return &p, err
}

func (m *ProductModel) Create(p *Product) error {
    if err := p.Validate(); err != nil { return err }
    p.CreatedAt = time.Now()
    return m.DB.QueryRow(
        `INSERT INTO products(name,description,price,stock,created_at) VALUES($1,$2,$3,$4,$5) RETURNING id`,
        p.Name, p.Description, p.Price, p.Stock, p.CreatedAt,
    ).Scan(&p.ID)
}
controller/product_controller.go — Controller (Logic + Routing)Go
package controller

import (
    "encoding/json"
    "html/template"
    "net/http"
    "strconv"
    "myapp/model"
)

// ── Controller: รับ HTTP request, เรียก Model, เลือก View ────────────────
// รู้จัก HTTP และ Model — แต่ไม่รู้จัก Database detail

type ProductController struct {
    model     *model.ProductModel
    templates *template.Template
}

func NewProductController(m *model.ProductModel, tmpl *template.Template) *ProductController {
    return &ProductController{model: m, templates: tmpl}
}

// Index — แสดงรายการสินค้าทั้งหมด
// GET /products
func (c *ProductController) Index(w http.ResponseWriter, r *http.Request) {
    products, err := c.model.All() // เรียก Model
    if err != nil {
        http.Error(w, "Failed to load products", http.StatusInternalServerError)
        return
    }

    // เลือก View ตาม Accept header
    if r.Header.Get("Accept") == "application/json" {
        // View: JSON
        w.Header().Set("Content-Type", "application/json")
        json.NewEncoder(w).Encode(products)
        return
    }

    // View: HTML template
    c.templates.ExecuteTemplate(w, "products/index.html", map[string]interface{}{
        "Title":    "สินค้าทั้งหมด",
        "Products": products,
    })
}

// Show — แสดงรายละเอียดสินค้า
// GET /products/{id}
func (c *ProductController) Show(w http.ResponseWriter, r *http.Request) {
    idStr := r.PathValue("id")
    id, err := strconv.Atoi(idStr)
    if err != nil {
        http.Error(w, "Invalid product ID", http.StatusBadRequest)
        return
    }

    product, err := c.model.ByID(id)
    if err != nil {
        http.Error(w, "Product not found", http.StatusNotFound)
        return
    }

    c.templates.ExecuteTemplate(w, "products/show.html", map[string]interface{}{
        "Product": product,
    })
}

// Create — สร้างสินค้าใหม่ (API)
// POST /products
func (c *ProductController) Create(w http.ResponseWriter, r *http.Request) {
    var input struct {
        Name        string  `json:"name"`
        Description string  `json:"description"`
        Price       float64 `json:"price"`
        Stock       int     `json:"stock"`
    }

    if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
        http.Error(w, "Invalid request", http.StatusBadRequest)
        return
    }

    product := &model.Product{
        Name: input.Name, Description: input.Description,
        Price: input.Price, Stock: input.Stock,
    }

    if err := c.model.Create(product); err != nil { // Business rule ใน Model
        http.Error(w, err.Error(), http.StatusBadRequest)
        return
    }

    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(http.StatusCreated)
    json.NewEncoder(w).Encode(product)
}

cmd/main.go — Wire Router + Controller + ModelGo
package main

import (
    "database/sql"
    "html/template"
    "log"
    "net/http"
    "myapp/controller"
    "myapp/model"
    _ "github.com/lib/pq"
)

func main() {
    // Setup
    db, _ := sql.Open("postgres", "postgres://localhost/shop")

    // Model (ชั้นข้อมูล)
    productModel := &model.ProductModel{DB: db}

    // View (templates)
    tmpl := template.Must(template.ParseGlob("view/templates/**/*.html"))

    // Controller (ตัวกลาง)
    productCtrl := controller.NewProductController(productModel, tmpl)

    // Router — bind URLs กับ Controller methods
    mux := http.NewServeMux()
    mux.HandleFunc("GET  /products",     productCtrl.Index)
    mux.HandleFunc("GET  /products/{id}", productCtrl.Show)
    mux.HandleFunc("POST /products",     productCtrl.Create)

    // Static files
    mux.Handle("/static/", http.StripPrefix("/static/", http.FileServer(http.Dir("view/static"))))

    log.Println("MVC App running on :8080")
    log.Fatal(http.ListenAndServe(":8080", mux))
}
✅ สรุป MVC
🗃️
Model — Rules
  • Business logic & validation
  • Database queries
  • ไม่รู้จัก HTTP หรือ HTML
  • Testable อิสระ
🖼️
View — Display
  • แสดงข้อมูลที่รับมาเท่านั้น
  • ไม่มี business logic
  • HTML Template / JSON
  • เปลี่ยน UI ไม่ต้องแก้ Model
🎮
Controller — Bridge
  • รับ Input → เรียก Model
  • ส่งข้อมูล → View
  • Thin Controller (ไม่อ้วน!)
  • ไม่ใส่ logic ซับซ้อนที่นี่
Anti-pattern: Fat Controller — Controller ที่มี business logic เยอะ ควรย้าย logic ไปอยู่ใน Model หรือ Service ทำให้ Controller "Thin" — แค่ coordinate เท่านั้น
