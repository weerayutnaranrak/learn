Saga Pattern

Distributed Transaction ใน Microservices — เมื่อ 2PC ไม่เหมาะอีกต่อไป Saga ใช้ compensating transactions แทนเพื่อรักษา consistency

❓ ปัญหา: Distributed Transaction
เมื่อ transaction ต้องข้ามหลาย service พร้อมกัน
สถานการณ์: ลูกค้าสั่งซื้อ → ต้อง 1) ตัดสต็อก 2) ตัดเงิน 3) สร้าง shipment — ถ้า service ใดล้มเหลวระหว่างทาง ข้อมูลจะ inconsistent
2-Phase Commit (2PC) แก้ได้ไหม? ได้ใน monolith แต่ใน microservices — 2PC ต้องการ distributed lock ข้ามหลาย DB ซึ่งทำให้ latency สูงและ availability ต่ำมาก
Happy path: ทุก step สำเร็จ
1. Create Order
→
2. Reserve Stock
→
3. Charge Payment
→
4. Create Shipment
→
✅ Done
Failure path: Payment ล้มเหลว — ต้อง compensate
1. Create Order ✓
→
2. Reserve Stock ✓
→
3. Payment ✗
→
↩ Release Stock
→
↩ Cancel Order
Saga = Eventual Consistency: ไม่ได้รับประกัน ACID atomicity ทั้งหมดในทันที แต่รับประกันว่าระบบจะ สุดท้ายกลับสู่ consistent state ไม่ว่า step ไหนจะล้มเหลว
🎭 Choreography-based Saga
แต่ละ service ฟัง event และตัดสินใจเอง — ไม่มี central coordinator
วิธีทำงาน: แต่ละ service publish event หลังทำงานสำเร็จ → service ถัดไปฟัง event นั้นแล้วทำงานต่อ → ถ้า error ก็ publish compensating event
เหมาะกับ: saga ที่ไม่ซับซ้อน, step น้อย, team อยากให้แต่ละ service เป็นอิสระ
Choreography Saga — Order flow ผ่าน events Go · Choreography 
// ── Events ────────────────────────────────────────────────
type OrderPlaced    struct{ OrderID, UserID string; Items []Item }
type StockReserved  struct{ OrderID string }
type StockFailed    struct{ OrderID, Reason string }
type PaymentCharged struct{ OrderID string; Amount float64 }
type PaymentFailed  struct{ OrderID, Reason string }
type OrderConfirmed struct{ OrderID string }
type OrderCancelled struct{ OrderID, Reason string }

// ── Order Service ─────────────────────────────────────────
type OrderService struct {
    repo OrderRepo
    bus  EventBus
}

func (s *OrderService) PlaceOrder(cmd PlaceOrderCmd) error {
    order := NewOrder(cmd.UserID, cmd.Items)
    order.Status = "pending"
    s.repo.Save(order)

    // publish แล้วรอ event ตอบกลับ
    s.bus.Publish("order.placed", OrderPlaced{
        OrderID: order.ID, UserID: order.UserID, Items: order.Items,
    })
    return nil
}

// ฟัง compensating event จาก downstream
func (s *OrderService) OnPaymentFailed(e PaymentFailed) {
    order, _ := s.repo.FindByID(e.OrderID)
    order.Status = "cancelled"
    s.repo.Save(order)
    s.bus.Publish("order.cancelled", OrderCancelled{
        OrderID: e.OrderID, Reason: e.Reason,
    })
}

// ── Inventory Service ─────────────────────────────────────
type InventoryService struct {
    repo InventoryRepo
    bus  EventBus
}

// ฟัง order.placed → จอง stock
func (s *InventoryService) OnOrderPlaced(e OrderPlaced) {
    ok, err := s.repo.Reserve(e.Items)
    if err != nil || !ok {
        s.bus.Publish("stock.failed", StockFailed{
            OrderID: e.OrderID, Reason: "out of stock",
        })
        return
    }
    s.bus.Publish("stock.reserved", StockReserved{OrderID: e.OrderID})
}

// ฟัง compensating event — คืน stock
func (s *InventoryService) OnOrderCancelled(e OrderCancelled) {
    s.repo.Release(e.OrderID) // compensating transaction
}

// ── Payment Service ───────────────────────────────────────
type PaymentService struct {
    gateway PaymentGateway
    bus     EventBus
}

// ฟัง stock.reserved → เก็บเงิน
func (s *PaymentService) OnStockReserved(e StockReserved) {
    order, _ := getOrderAmount(e.OrderID)
    if err := s.gateway.Charge(order.UserID, order.Total); err != nil {
        s.bus.Publish("payment.failed", PaymentFailed{
            OrderID: e.OrderID, Reason: err.Error(),
        })
        return
    }
    s.bus.Publish("payment.charged", PaymentCharged{
        OrderID: e.OrderID, Amount: order.Total,
    })
}

// ── Wire-up subscriptions ─────────────────────────────────
func wireChoreography(bus EventBus, inv *InventoryService,
    pay *PaymentService, ord *OrderService) {

    bus.Subscribe("order.placed",     inv.OnOrderPlaced)
    bus.Subscribe("stock.reserved",   pay.OnStockReserved)
    bus.Subscribe("payment.failed",   inv.OnOrderCancelled)
    bus.Subscribe("payment.failed",   ord.OnPaymentFailed)
    bus.Subscribe("order.cancelled",  inv.OnOrderCancelled)
}
🎬 Orchestration-based Saga
มี central orchestrator ที่รู้ทุก step และสั่งแต่ละ service
วิธีทำงาน: Orchestrator (state machine) สั่ง service แต่ละตัวตามลำดับ ถ้า step ใด fail orchestrator สั่ง compensate ตาม step ย้อนหลัง
เหมาะกับ: saga ที่ซับซ้อน มีหลาย branch, ต้องการเห็น flow ชัดเจนจากจุดเดียว, ง่ายต่อการ debug
Orchestration Saga — state machine coordinator Go · Orchestration 
type SagaState string

const (
    StateStarted          SagaState = "started"
    StateStockReserved    SagaState = "stock_reserved"
    StatePaymentCharged   SagaState = "payment_charged"
    StateCompleted        SagaState = "completed"
    StateCompensating     SagaState = "compensating"
    StateFailed           SagaState = "failed"
)

type OrderSaga struct {
    ID        string
    OrderID   string
    State     SagaState
    inventory InventoryService
    payment   PaymentService
    order     OrderService
    repo      SagaRepo // persist saga state
}

// Execute: ประมวลผล step ถัดไปตาม state ปัจจุบัน
func (s *OrderSaga) Execute(ctx context.Context) error {
    for {
        switch s.State {
        case StateStarted:
            if err := s.reserveStock(ctx); err != nil {
                return s.startCompensation(ctx, err)
            }
            s.setState(StateStockReserved)

        case StateStockReserved:
            if err := s.chargePayment(ctx); err != nil {
                return s.startCompensation(ctx, err)
            }
            s.setState(StatePaymentCharged)

        case StatePaymentCharged:
            s.order.Confirm(ctx, s.OrderID)
            s.setState(StateCompleted)
            return nil

        case StateCompensating:
            s.compensate(ctx)
            s.setState(StateFailed)
            return fmt.Errorf("saga failed, compensation done")

        case StateCompleted, StateFailed:
            return nil
        }
    }
}

func (s *OrderSaga) reserveStock(ctx context.Context) error {
    return s.inventory.Reserve(ctx, s.OrderID)
}

func (s *OrderSaga) chargePayment(ctx context.Context) error {
    return s.payment.Charge(ctx, s.OrderID)
}

// Compensate ย้อนกลับตาม state ที่ถึงแล้ว
func (s *OrderSaga) compensate(ctx context.Context) {
    switch s.State {
    case StateStockReserved, StatePaymentCharged:
        // stock ถูก reserve แล้ว → ต้อง release
        s.inventory.Release(ctx, s.OrderID)
        fallthrough
    case StateStarted:
        s.order.Cancel(ctx, s.OrderID)
    }
}

func (s *OrderSaga) startCompensation(ctx context.Context, err error) error {
    s.setState(StateCompensating)
    return s.Execute(ctx) // re-enter loop → เข้า case Compensating
}

func (s *OrderSaga) setState(state SagaState) {
    s.State = state
    s.repo.Save(s) // persist state ทุกครั้ง → resume ได้หาก crash
}

// ── Saga Manager: สร้างและรัน saga ──────────────────────────
type SagaManager struct {
    repo      SagaRepo
    inventory InventoryService
    payment   PaymentService
    order     OrderService
}

func (m *SagaManager) StartOrderSaga(ctx context.Context, orderID string) error {
    saga := &OrderSaga{
        ID: uuid.New().String(), OrderID: orderID,
        State:     StateStarted,
        inventory: m.inventory,
        payment:   m.payment,
        order:     m.order,
        repo:      m.repo,
    }
    m.repo.Save(saga)
    return saga.Execute(ctx)
}

// Resume saga ที่ค้างจาก crash — idempotent
func (m *SagaManager) Resume(ctx context.Context, sagaID string) error {
    saga, _ := m.repo.FindByID(sagaID)
    saga.inventory = m.inventory // re-inject deps
    saga.payment   = m.payment
    saga.order     = m.order
    saga.repo      = m.repo
    return saga.Execute(ctx)
}
⚖️ Choreography vs Orchestration
เลือกใช้ให้ถูกกับ use case
มิติ
🎭 Choreography
🎬 Orchestration
Coupling
Loose — services ไม่รู้จักกัน
Orchestrator รู้จักทุก service
Visibility
ยาก — ต้อง trace events
ง่าย — orchestrator เก็บ state
Complexity
ง่ายเริ่มต้น ซับซ้อนเมื่อ flow เพิ่ม
ซับซ้อนกว่าตอนเริ่ม แต่ scale ได้
Testing
ยาก — ต้องจำลอง event chain
ง่าย — test orchestrator โดยตรง
เหมาะกับ
Simple flow, Event-driven first system
Complex flow, Long-running process
Idempotency สำคัญมาก: ทุก step ใน saga ต้องเป็น idempotent — รัน 2 ครั้งให้ผลเหมือนรัน 1 ครั้ง เพราะ message อาจถูกส่งซ้ำ ใช้ idempotency key ทุกครั้ง
📤 Outbox Pattern — ส่ง event ได้แน่นอน
แก้ปัญหา "save to DB แล้ว publish event ล้มเหลว"
Transactional Outbox — atomic save + publish Go · Outbox 
// ปัญหา: save order สำเร็จ แต่ publish event ล้มเหลว
// → order ถูก create แต่ inventory ไม่รู้ → inconsistent
func (s *OrderService) PlaceOrderBroken(o Order) error {
    s.repo.Save(o)                    // ✓ saved
    s.bus.Publish("order.placed", o)  // ✗ network error → event lost!
    return nil
}

// ── Outbox: บันทึก event ใน DB transaction เดียวกับ data ──
type OutboxEvent struct {
    ID        string
    Topic     string
    Payload   []byte
    CreatedAt time.Time
    Published bool
}

func (s *OrderService) PlaceOrder(ctx context.Context, o Order) error {
    return s.db.WithTransaction(ctx, func(tx *sql.Tx) error {
        // 1. save order
        if err := s.repo.SaveTx(tx, o); err != nil {
            return err
        }

        // 2. save event ใน outbox table — transaction เดียวกัน
        payload, _ := json.Marshal(OrderPlaced{OrderID: o.ID})
        event := OutboxEvent{
            ID: uuid.New().String(), Topic: "order.placed",
            Payload: payload, CreatedAt: time.Now(),
        }
        return s.outboxRepo.SaveTx(tx, event)
        // ถ้า step ใดล้มเหลว → rollback ทั้ง 2 → consistent
    })
}

// Outbox Relay: process แยก poll outbox แล้วส่งจริง
type OutboxRelay struct {
    repo   OutboxRepo
    bus    EventBus
    ticker *time.Ticker
}

func (r *OutboxRelay) Run(ctx context.Context) {
    for {
        select {
        case <-r.ticker.C:
            events, _ := r.repo.FindUnpublished(ctx, 100)
            for _, e := range events {
                if err := r.bus.Publish(e.Topic, e.Payload); err != nil {
                    continue // retry next tick
                }
                r.repo.MarkPublished(ctx, e.ID) // mark สำเร็จ
            }
        case <-ctx.Done():
            return
        }
    }
}
Outbox + Saga = Reliable Saga: Outbox รับประกันว่า event ถึง message broker แน่นอน, Saga รับประกันว่าระบบกลับสู่ consistent state — ใช้ร่วมกันสำหรับ distributed transaction ที่เชื่อถือได้
