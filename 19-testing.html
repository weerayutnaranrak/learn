// ── Interface ที่จะ test ───────────────────────────────────
type UserRepository interface {
    FindByID(id string) (*User, error)
    Save(u *User) error
    CountByEmail(email string) (int, error)
}

// ── STUB: คืนค่าที่กำหนดล่วงหน้า ─────────────────────────
type StubUserRepo struct {
    user *User
    err  error
}

func (s *StubUserRepo) FindByID(id string) (*User, error) {
    return s.user, s.err // คืนสิ่งที่กำหนดไว้เสมอ
}
func (s *StubUserRepo) Save(u *User) error           { return nil }
func (s *StubUserRepo) CountByEmail(e string) (int, error) { return 0, nil }

// ── MOCK: บันทึก calls แล้ว assert ─────────────────────────
type MockUserRepo struct {
    SaveCalls []User    // บันทึก arguments ที่ถูกส่งมา
    FindCalls []string
}

func (m *MockUserRepo) FindByID(id string) (*User, error) {
    m.FindCalls = append(m.FindCalls, id)
    return &User{ID: id}, nil
}
func (m *MockUserRepo) Save(u *User) error {
    m.SaveCalls = append(m.SaveCalls, *u)
    return nil
}
func (m *MockUserRepo) CountByEmail(e string) (int, error) { return 0, nil }

// ── FAKE: in-memory implementation จริงๆ ───────────────────
type InMemoryUserRepo struct {
    data map[string]*User
}

func NewInMemoryUserRepo() *InMemoryUserRepo {
    return &InMemoryUserRepo{data: make(map[string]*User)}
}
func (r *InMemoryUserRepo) FindByID(id string) (*User, error) {
    u, ok := r.data[id]
    if !ok { return nil, errors.New("not found") }
    return u, nil
}
func (r *InMemoryUserRepo) Save(u *User) error {
    r.data[u.ID] = u
    return nil
}
func (r *InMemoryUserRepo) CountByEmail(e string) (int, error) {
    count := 0
    for _, u := range r.data {
        if u.Email == e { count++ }
    }
    return count, nil
}

// ── Tests ──────────────────────────────────────────────────
func TestRegisterUser_EmailAlreadyExists(t *testing.T) {
    // Stub: ทำให้ CountByEmail คืน 1 (มี email แล้ว)
    repo := &StubUserRepo{}
    // ใช้ Fake แทน เพราะ logic ซับซ้อน
    fakeRepo := NewInMemoryUserRepo()
    fakeRepo.Save(&User{ID: "1", Email: "a@test.com"})

    svc := NewUserService(fakeRepo)
    err := svc.Register("a@test.com", "pass")

    if err == nil {
        t.Error("expected duplicate email error")
    }
    _ = repo // unused here
}

func TestRegisterUser_CallsSave(t *testing.T) {
    // Mock: ตรวจว่า Save ถูกเรียก
    mock := &MockUserRepo{}
    svc := NewUserService(mock)

    svc.Register("new@test.com", "pass")

    if len(mock.SaveCalls) != 1 {
        t.Errorf("expected Save called once, got %d", len(mock.SaveCalls))
    }
    if mock.SaveCalls[0].Email != "new@test.com" {
        t.Error("saved wrong email")
    }
}
// ── Interface ที่จะ test ───────────────────────────────────
interface User { id: string; email: string }

interface UserRepository {
  findByID(id: string): Promise;
  save(u: User): Promise;
  countByEmail(email: string): Promise;
}

// ── STUB: คืนค่าที่กำหนดล่วงหน้า ─────────────────────────
const stubUserRepo: UserRepository = {
  findByID: async (_id) => ({ id: '1', email: 'a@test.com' }),
  save: async (_u) => {},
  countByEmail: async (_e) => 0,
};

// ── MOCK: บันทึก calls แล้ว assert (Vitest / Jest) ─────────
import { vi, describe, it, expect } from 'vitest';

const mockUserRepo: UserRepository = {
  findByID: vi.fn().mockResolvedValue({ id: '1', email: 'a@test.com' }),
  save:     vi.fn().mockResolvedValue(undefined),
  countByEmail: vi.fn().mockResolvedValue(0),
};

// ── FAKE: in-memory implementation จริงๆ ───────────────────
class InMemoryUserRepo implements UserRepository {
  private data = new Map();

  async findByID(id: string) {
    return this.data.get(id) ?? null;
  }
  async save(u: User) {
    this.data.set(u.id, u);
  }
  async countByEmail(email: string) {
    let count = 0;
    for (const u of this.data.values()) {
      if (u.email === email) count++;
    }
    return count;
  }
}

// ── Tests ──────────────────────────────────────────────────
describe('UserService', () => {
  it('throws when email already exists', async () => {
    const fakeRepo = new InMemoryUserRepo();
    await fakeRepo.save({ id: '1', email: 'a@test.com' });

    const svc = new UserService(fakeRepo);
    await expect(svc.register('a@test.com', 'pass'))
      .rejects.toThrow('duplicate email');
  });

  it('calls save with correct email', async () => {
    const mock = { ...mockUserRepo };
    (mock.save as ReturnType).mockClear();
    (mock.countByEmail as ReturnType).mockResolvedValue(0);

    const svc = new UserService(mock);
    await svc.register('new@test.com', 'pass');

    expect(mock.save).toHaveBeenCalledTimes(1);
    expect((mock.save as ReturnType).mock.calls[0][0].email)
      .toBe('new@test.com');
  });
});
📋 Table-driven Tests
A Go idiom for testing many cases without duplicating codeGo idiom ที่ทำให้ test เยอะ case ได้โดยไม่ซ้ำ code
Idea: write the test logic once but run it against many input sets using a []struct as a test table — Go testing supports this with t.Run()
Benefits: add a new case by just adding a row, clear test names, easy to debug on failureแนวคิด: เขียน test logic ครั้งเดียว แต่รันกับ input หลายชุดโดยใช้ []struct เป็น test table — Go testing รองรับด้วย t.Run()
ข้อดี: เพิ่ม case ใหม่แค่เพิ่มแถวใน table, ชื่อ test ชัดเจน, ง่ายต่อการ debug เมื่อ fail
❌ A separate test per case — repetitiveเขียน test แยกทุก case — ซ้ำซ้อน anti-patternผิดหลักการ

func TestCalcDiscount(t *testing.T) {
    tests := []struct {
        name     string
        total    float64
        userType string
        want     float64
    }{
        {"vip gets 20%",     1000, "vip",      200},
        {"member gets 10%",  1000, "member",   100},
        {"new user gets 5%", 1000, "new_user",  50},
        {"unknown gets 0",   1000, "guest",      0},
        {"zero total",          0, "vip",         0},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            got := CalcDiscount(tt.total, tt.userType)
            if got != tt.want {
                t.Errorf("got %v, want %v", got, tt.want)
            }
        })
    }
}

// ── 1. RED: เขียน test ก่อน — ยังไม่มี ValidatePassword ─────
func TestValidatePassword(t *testing.T) {
    tests := []struct {
        name    string
        pwd     string
        wantErr bool
    }{
        {"too short",          "abc",       true},
        {"no number",          "abcdefgh",  true},
        {"no uppercase",       "abcdef1",   true},
        {"valid password",     "Abcdef1!",  false},
    }
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            err := ValidatePassword(tt.pwd)
            if (err != nil) != tt.wantErr {
                t.Errorf("ValidatePassword(%q) err=%v, wantErr=%v",
                    tt.pwd, err, tt.wantErr)
            }
        })
    }
}

// ── 2. GREEN: เขียน code น้อยที่สุดให้ผ่าน ─────────────────
func ValidatePassword(pwd string) error {
    if len(pwd) < 8 {
        return errors.New("password too short")
    }
    hasUpper, hasNum := false, false
    for _, c := range pwd {
        if unicode.IsUpper(c) { hasUpper = true }
        if unicode.IsDigit(c) { hasNum = true }
    }
    if !hasUpper { return errors.New("must contain uppercase") }
    if !hasNum   { return errors.New("must contain number") }
    return nil
}

// ── 3. REFACTOR: ทำให้ดีขึ้นโดย test ยังผ่าน ────────────────
type PasswordRule func(string) error

func MinLength(n int) PasswordRule {
    return func(pwd string) error {
        if len(pwd) < n {
            return fmt.Errorf("password must be at least %d characters", n)
        }
        return nil
    }
}

func HasUppercase() PasswordRule {
    return func(pwd string) error {
        for _, c := range pwd {
            if unicode.IsUpper(c) { return nil }
        }
        return errors.New("password must contain uppercase")
    }
}

func ValidatePassword(pwd string, rules ...PasswordRule) error {
    for _, rule := range rules {
        if err := rule(pwd); err != nil { return err }
    }
    return nil
}

🔌 Integration Tests
Test several components working together against real infraทดสอบ component หลายตัวทำงานร่วมกันกับ infra จริง
When to use: when you need confidence that a SQL query is correct, a migration doesn't break, or an HTTP handler returns a correctly formatted response
In Go: use testcontainers-go to spin up a real Docker container in tests, or httptest.NewServer() for HTTP integration testsเมื่อไหร่ใช้: เมื่อต้องการมั่นใจว่า SQL query ถูกต้อง, migration ไม่พัง, หรือ HTTP handler คืน response ถูก format

// httptest.NewRecorder() — ไม่ต้อง start server จริง
func TestOrderHandler_PlaceOrder(t *testing.T) {
    // Fake repo แทน Postgres
    repo := NewInMemoryOrderRepo()
    svc  := NewOrderService(repo, &NoopNotifier{})
    h    := NewOrderHandler(svc)

    // สร้าง HTTP request จริง
    body := `{"user_id":"u1","items":[{"sku":"A","qty":2}]}`
    req  := httptest.NewRequest("POST", "/orders", strings.NewReader(body))
    req.Header.Set("Content-Type", "application/json")
    w := httptest.NewRecorder()

    h.PlaceOrder(w, req)

    res := w.Result()
    if res.StatusCode != http.StatusCreated {
        t.Errorf("status=%d want 201", res.StatusCode)
    }

    var resp OrderResponse
    json.NewDecoder(res.Body).Decode(&resp)
    if resp.OrderID == "" {
        t.Error("expected order ID in response")
    }

    // verify side effect: order actually saved
    saved, err := repo.FindByID(resp.OrderID)
    if err != nil || saved == nil {
        t.Error("order was not saved")
    }
}

func TestOrderRepo_Integration(t *testing.T) {
    if testing.Short() { t.Skip("skipping integration test") }

🎭 How to pick a test doubleTest Double เลือกอย่างไร
Stub: you need controllable input
Mock: you must verify a method was called
Fake: logic is complex, mocking gets messyStub: ต้องการ input ที่ควบคุมได้
Mock: ต้อง verify ว่า method ถูกเรียก
Fake: logic ซับซ้อน, mock จะยุ่งยาก
Avoid over-mocking: if there's more mock than real code, the test may not test real behavior but only that "the code calls methods in a certain order" — which is very brittle. Use a Fake instead when possibleหลีกเลี่ยง Over-mocking: ถ้า mock เยอะกว่า code จริง — test นั้นอาจไม่ได้ test behavior จริง แต่ test ว่า "code เรียก method ในลำดับที่ต้องการ" ซึ่งเปราะบางมาก ใช้ Fake แทนถ้าเป็นไปได้
