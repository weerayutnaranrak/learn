Event Driven Design

ออกแบบระบบที่สื่อสารกันผ่าน Events — ลด Coupling ระหว่าง Services ทำให้ Scale ได้ง่าย และทนต่อ Failure

💡 Event Driven คืออะไร?
Pattern ที่ Components สื่อสารกันผ่าน Events แทนการ call กันโดยตรง

แทนที่ Service A จะ call Service B โดยตรง (Tight Coupling) — Service A จะ publish event ว่า "มีอะไรเกิดขึ้น" แล้ว Service B, C, D จะ subscribe event นั้นเอง

events/bus.go — Event Bus (In-Memory Implementation) Go 
package events

import (
    "context"
    "fmt"
    "log"
    "sync"
    "time"
)

// Bus — interface ของ Event Bus
type Bus interface {
    Subscribe(eventType string, handler Handler)
    Publish(ctx context.Context, event Event) error
    PublishAll(ctx context.Context, events []Event) error
}

// InMemoryBus — Synchronous bus สำหรับ development/testing
// สำหรับ production ใช้ Kafka/RabbitMQ/NATS แทน
type InMemoryBus struct {
    mu       sync.RWMutex
    handlers map[string][]Handler
}

func NewInMemoryBus() *InMemoryBus {
    return &InMemoryBus{handlers: make(map[string][]Handler)}
}

func (b *InMemoryBus) Subscribe(eventType string, handler Handler) {
    b.mu.Lock()
    defer b.mu.Unlock()
    b.handlers[eventType] = append(b.handlers[eventType], handler)
    log.Printf("[Bus] Subscribed to %s", eventType)
}

func (b *InMemoryBus) Publish(ctx context.Context, event Event) error {
    b.mu.RLock()
    handlers := append([]Handler{}, b.handlers[event.Type]...) // copy slice
    b.mu.RUnlock()

    if len(handlers) == 0 {
        log.Printf("[Bus] No subscribers for %s", event.Type)
        return nil
    }

    var wg sync.WaitGroup
    errCh := make(chan error, len(handlers))

    for _, h := range handlers {
        wg.Add(1)
        go func(handler Handler) {
            defer wg.Done()
            // Retry 3 ครั้ง ถ้า handler fail
            if err := retryHandler(ctx, handler, event, 3); err != nil {
                errCh <- fmt.Errorf("handler for %s: %w", event.Type, err)
            }
        }(h)
    }

    wg.Wait()
    close(errCh)

    var errs []error
    for err := range errCh {
        errs = append(errs, err)
    }
    if len(errs) > 0 {
        return fmt.Errorf("some handlers failed: %v", errs)
    }
    return nil
}

func (b *InMemoryBus) PublishAll(ctx context.Context, events []Event) error {
    for _, e := range events {
        if err := b.Publish(ctx, e); err != nil {
            return err
        }
    }
    return nil
}

// retryHandler — retry ด้วย exponential backoff
func retryHandler(ctx context.Context, h Handler, event Event, maxRetries int) error {
    var err error
    for attempt := 0; attempt < maxRetries; attempt++ {
        if attempt > 0 {
            wait := time.Duration(attempt*attempt) * 100 * time.Millisecond
            log.Printf("[Bus] Retry %d for %s after %v", attempt, event.Type, wait)
            select {
            case <-time.After(wait):
            case <-ctx.Done():
                return ctx.Err()
            }
        }
        if err = h(ctx, event); err == nil {
            return nil
        }
    }
    return fmt.Errorf("after %d retries: %w", maxRetries, err)
}
// events/bus.ts — In-Memory Event Bus

import { DomainEvent, Handler } from './event';

// EventBus interface
export interface EventBus {
  subscribe(eventType: string, handler: Handler): void;
  publish(event: DomainEvent): Promise;
  publishAll(events: DomainEvent[]): Promise;
}

// InMemoryBus — for development/testing
// For production use Kafka / RabbitMQ / NATS adapter
export class InMemoryBus implements EventBus {
  private handlers = new Map>>();

  subscribe(eventType: string, handler: Handler): void {
    if (!this.handlers.has(eventType)) {
      this.handlers.set(eventType, new Set());
    }
    this.handlers.get(eventType)!.add(handler as Handler);
    console.log(`[Bus] Subscribed to ${eventType}`);
  }

  async publish(event: DomainEvent): Promise {
    const handlers = this.handlers.get(event.type);
    if (!handlers || handlers.size === 0) {
      console.log(`[Bus] No subscribers for ${event.type}`);
      return;
    }

    // Run all handlers in parallel — equivalent to goroutines + WaitGroup
    const results = await Promise.allSettled(
      [...handlers].map(handler => retryHandler(handler, event, 3))
    );

    const failures = results
      .filter((r): r is PromiseRejectedResult => r.status === 'rejected')
      .map(r => r.reason as Error);

    if (failures.length > 0) {
      throw new Error(`Some handlers failed: ${failures.map(e => e.message).join(', ')}`);
    }
  }

  async publishAll(events: DomainEvent[]): Promise {
    for (const event of events) {
      await this.publish(event);
    }
  }
}

// retryHandler — exponential backoff retry
async function retryHandler(
  handler: Handler,
  event: DomainEvent,
  maxRetries: number
): Promise {
  let lastError: unknown;
  for (let attempt = 0; attempt < maxRetries; attempt++) {
    if (attempt > 0) {
      const waitMs = attempt * attempt * 100; // exponential backoff
      console.log(`[Bus] Retry ${attempt} for ${event.type} after ${waitMs}ms`);
      await new Promise(resolve => setTimeout(resolve, waitMs));
    }
    try {
      await handler(event);
      return; // success
    } catch (err) {
      lastError = err;
    }
  }
  throw new Error(`After ${maxRetries} retries: ${(lastError as Error)?.message}`);
}
services/order_service.go — Publisher Go 
package services

import (
    "context"
    "fmt"
    "myapp/events"
)

// OrderService — publishes events เมื่อสิ่งเกิดขึ้น
type OrderService struct {
    repo orderRepo
    bus  events.Bus
}

func NewOrderService(repo orderRepo, bus events.Bus) *OrderService {
    return &OrderService{repo: repo, bus: bus}
}

type PlaceOrderRequest struct {
    CustomerID string
    Items      []OrderItem
}

func (s *OrderService) PlaceOrder(ctx context.Context, req PlaceOrderRequest) (*Order, error) {
    // 1. Business logic
    order, err := s.repo.Create(ctx, req)
    if err != nil {
        return nil, fmt.Errorf("create order: %w", err)
    }

    // 2. Publish event — บอกโลกว่า "order ถูกสร้างแล้ว"
    event, err := events.NewEvent(events.EventOrderPlaced, events.OrderPlacedPayload{
        OrderID:    order.ID,
        CustomerID: req.CustomerID,
        Total:      order.Total,
        Items:      len(req.Items),
    })
    if err != nil {
        return nil, fmt.Errorf("create event: %w", err)
    }

    // Fire — ไม่รอ response ว่า subscribers ทำอะไร
    if err := s.bus.Publish(ctx, event); err != nil {
        // Log แต่ไม่ fail — order สร้างแล้ว
        // (หรือใช้ Outbox Pattern สำหรับ at-least-once delivery)
        log.Printf("warn: publish event failed: %v", err)
    }

    return order, nil
}

func (s *OrderService) ConfirmOrder(ctx context.Context, orderID string) error {
    order, err := s.repo.FindByID(ctx, orderID)
    if err != nil {
        return err
    }
    if err := order.Confirm(); err != nil {
        return err
    }
    if err := s.repo.Save(ctx, order); err != nil {
        return err
    }

    // Publish ว่า confirmed แล้ว
    event, _ := events.NewEvent(events.EventOrderConfirmed, map[string]string{
        "order_id": orderID,
    })
    return s.bus.Publish(ctx, event)
}
// services/order.service.ts — Publisher

import { EventBus } from '../events/bus';
import { newEvent, EventTypes, OrderPlacedPayload } from '../events/event';

export interface PlaceOrderRequest {
  customerId: string;
  items: OrderItem[];
}

export interface OrderItem {
  productId: string;
  quantity: number;
  price: number;
}

export interface Order {
  id: string;
  customerId: string;
  total: number;
  status: string;
}

// OrderService — publishes events when things happen
export class OrderService {
  constructor(
    private readonly repo: OrderRepository,
    private readonly bus: EventBus,
  ) {}

  async placeOrder(req: PlaceOrderRequest): Promise {
    // 1. Business logic — create the order
    const order = await this.repo.create(req);

    // 2. Publish event — tell the world "an order was placed"
    const event = newEvent(EventTypes.OrderPlaced, {
      orderId:    order.id,
      customerId: req.customerId,
      total:      order.total,
      items:      req.items.length,
    });

    // Fire-and-forget — don't fail the order if publish fails
    // (use Outbox Pattern for at-least-once delivery in production)
    this.bus.publish(event).catch(err => {
      console.warn('warn: publish event failed:', err);
    });

    return order;
  }

  async confirmOrder(orderId: string): Promise {
    const order = await this.repo.findById(orderId);
    order.confirm(); // throws if not in 'pending' state
    await this.repo.save(order);

    // Publish that the order is confirmed
    const event = newEvent(EventTypes.OrderConfirmed, { orderId });
    await this.bus.publish(event);
  }
}

// Repository interface (dependency inversion)
interface OrderRepository {
  create(req: PlaceOrderRequest): Promise;
  findById(id: string): Promise;
  save(order: Order): Promise;
}

event_sourcing/order_aggregate.go — Event Sourced Aggregate Go 
package eventsourcing

import (
    "encoding/json"
    "time"
)

// OrderAggregate — State ถูก rebuild จาก Events
type OrderAggregate struct {
    ID         string
    CustomerID string
    Items      []Item
    Status     string
    Total      float64
    Version    int // Optimistic concurrency control

    uncommitted []StoredEvent // Events ที่ยังไม่ได้ persist
}

// StoredEvent — Event ที่ persist ใน Event Store
type StoredEvent struct {
    AggregateID string
    EventType   string
    Data        json.RawMessage
    OccurredAt  time.Time
    Version     int
}

// ── Apply Methods — rebuild state จาก events ──────────────────────────────────

// Apply — dispatch event ไปยัง handler ที่เหมาะสม
func (o *OrderAggregate) Apply(event StoredEvent) {
    switch event.EventType {
    case "OrderCreated":
        var e struct{ CustomerID string }
        json.Unmarshal(event.Data, &e)
        o.CustomerID = e.CustomerID
        o.Status = "pending"

    case "ItemAdded":
        var e struct{ Name string; Price float64; Qty int }
        json.Unmarshal(event.Data, &e)
        o.Items = append(o.Items, Item{Name: e.Name, Price: e.Price, Qty: e.Qty})
        o.Total += e.Price * float64(e.Qty)

    case "ItemRemoved":
        var e struct{ Name string; Price float64; Qty int }
        json.Unmarshal(event.Data, &e)
        o.Total -= e.Price * float64(e.Qty)
        // remove from items...

    case "OrderConfirmed":
        o.Status = "confirmed"

    case "OrderShipped":
        o.Status = "shipped"
    }
    o.Version = event.Version
}

// ── Command Methods — produce events ──────────────────────────────────────────

func (o *OrderAggregate) AddItem(name string, price float64, qty int) {
    data, _ := json.Marshal(map[string]interface{}{
        "name": name, "price": price, "qty": qty,
    })
    o.raise(StoredEvent{
        AggregateID: o.ID,
        EventType:   "ItemAdded",
        Data:        data,
        OccurredAt:  time.Now(),
        Version:     o.Version + 1,
    })
}

func (o *OrderAggregate) Confirm() error {
    if o.Status != "pending" {
        return fmt.Errorf("cannot confirm %s order", o.Status)
    }
    data, _ := json.Marshal(map[string]interface{}{"order_id": o.ID})
    o.raise(StoredEvent{
        AggregateID: o.ID,
        EventType:   "OrderConfirmed",
        Data:        data,
        OccurredAt:  time.Now(),
        Version:     o.Version + 1,
    })
    return nil
}

func (o *OrderAggregate) raise(event StoredEvent) {
    o.Apply(event)           // อัพเดท in-memory state ทันที
    o.uncommitted = append(o.uncommitted, event)
}

func (o *OrderAggregate) PopUncommitted() []StoredEvent {
    evts := o.uncommitted
    o.uncommitted = nil
    return evts
}

// ── Rebuild from history ────────────────────────────────────────────────────

// Reconstitute — สร้าง aggregate จาก event history
func Reconstitute(id string, events []StoredEvent) *OrderAggregate {
    agg := &OrderAggregate{ID: id}
    for _, e := range events {
        agg.Apply(e)
    }
    return agg
}
📊 CQRS — Command Query Responsibility Segregation
แยก Write Model (Command) ออกจาก Read Model (Query)

Write Side (Command) — ทำ business logic, validate, save events
Read Side (Query) — Denormalized views สำหรับ query เร็วๆ (ไม่ต้อง join หลายตาราง)

CQRS Flow
เมื่อไหร่ควรใช้ CQRS: เมื่อ Read และ Write มีความต้องการต่างกันมาก เช่น Write ต้องการ consistency สูง แต่ Read ต้องการ performance สูง

ไม่ต้องใช้ CQRS: สำหรับ CRUD ธรรมดา — เพิ่ม complexity โดยไม่จำเป็น
✅ สรุปและเมื่อไหร่ควรใช้
ควรใช้เมื่อ

เหมาะกับ

• Microservices ที่ต้องการ decoupling
• Systems ที่มีหลาย consumers สำหรับ event เดียว
• Audit trail / ประวัติย้อนหลัง
• Async processing ที่ tolerates delay

ระวัง

ความท้าทาย

• Eventual Consistency — ข้อมูลอาจไม่ sync ทันที
• Event Schema versioning — เปลี่ยน event structure ยาก
• Debugging — trace events ข้าม services
• Message ordering ต้องดูแล

Production Tools:
• Kafka — High throughput, persistent, replay ได้
• RabbitMQ — Feature rich routing, exchange patterns
• NATS — Ultra low latency, Go-native
• Google Pub/Sub — Managed, scalable, cloud-native
