Design Pattern #12

Repository Pattern

Separate business logic from data access with an abstraction layer — making it easy to test and easy to swap the databaseแยก Business Logic ออกจาก Data Access ด้วย Abstraction Layer — ทำให้ทดสอบง่าย เปลี่ยน Database ง่าย

What is the Repository Pattern?Repository Pattern คืออะไร?

The Repository Pattern puts an abstraction layer between business logic and data-access code. The business layer talks only to a Repository interface and doesn't know where the data is stored (PostgreSQL, MongoDB, in-memory, API)Repository Pattern เป็น design pattern ที่ใส่ abstraction layer ระหว่าง business logic กับ data access code โดย Business Layer จะคุยกับ Repository Interface เท่านั้น ไม่รู้จักว่า data เก็บอยู่ที่ไหน (PostgreSQL, MongoDB, In-Memory, API)

domain/order.go — Domain Entity
package domain

import (
	"time"
)

type OrderStatus string

const (
	StatusPending   OrderStatus = "pending"
	StatusPaid      OrderStatus = "paid"
	StatusCancelled OrderStatus = "cancelled"
)

type Order struct {
	ID         string
	CustomerID string
	Items      []OrderItem
	Total      float64
	Status     OrderStatus
	CreatedAt  time.Time
	ExpiresAt  time.Time
}

type OrderItem struct {
	ProductID string
	Quantity  int
	Price     float64
}

func (o *Order) IsExpired() bool {
	return time.Now().After(o.ExpiresAt)
}

func (o *Order) CanBeCancelled() bool {
	return o.Status == StatusPending && !o.IsExpired()
}
// domain/order.ts — Domain Entity

export type OrderStatus = 'pending' | 'paid' | 'cancelled';

export interface OrderItem {
  productId: string;
  quantity: number;
  price: number;
}

export class Order {
  constructor(
    public readonly id: string,
    public readonly customerId: string,
    public readonly items: OrderItem[],
    public total: number,
    public status: OrderStatus,
    public readonly createdAt: Date,
    public readonly expiresAt: Date,
  ) {}

  isExpired(): boolean {
    return new Date() > this.expiresAt;
  }

  canBeCancelled(): boolean {
    return this.status === 'pending' && !this.isExpired();
  }
}
domain/repository.go — Repository Interface (The Contract)
package domain

import "context"

// OrderRepository — the interface that business layer depends on.
// All implementations must satisfy this contract.
type OrderRepository interface {
	FindByID(ctx context.Context, id string) (*Order, error)
	FindByCustomerID(ctx context.Context, customerID string) ([]*Order, error)
	FindByStatus(ctx context.Context, status OrderStatus) ([]*Order, error)
	Save(ctx context.Context, order *Order) error
	Delete(ctx context.Context, id string) error
}

// Sentinel errors — defined in domain, not in infrastructure
var (
	ErrNotFound    = fmt.Errorf("order: not found")
	ErrDuplicate   = fmt.Errorf("order: duplicate")
)
// domain/repository.ts — Repository Interface (The Contract)
import { Order, OrderStatus } from './order';

// OrderRepository — the interface that business layer depends on.
// All implementations must satisfy this contract.
export interface OrderRepository {
  findById(id: string): Promise;
  findByCustomerId(customerId: string): Promise;
  findByStatus(status: OrderStatus): Promise;
  save(order: Order): Promise;
  delete(id: string): Promise;
}

// Domain errors — defined in domain, not in infrastructure
export class OrderNotFoundError extends Error {
  constructor(id: string) { super(`order: not found — ${id}`); }
}
export class OrderDuplicateError extends Error {
  constructor(id: string) { super(`order: duplicate — ${id}`); }
}
infrastructure/postgres_repo.go — PostgreSQL Implementation
package infrastructure

import (
	"context"
	"database/sql"
	"fmt"

	"myapp/domain"
)

type PostgresOrderRepo struct {
	db *sql.DB
}

func NewPostgresOrderRepo(db *sql.DB) domain.OrderRepository {
	return &PostgresOrderRepo{db: db}
}

func (r *PostgresOrderRepo) FindByID(ctx context.Context, id string) (*domain.Order, error) {
	row := r.db.QueryRowContext(ctx,
		`SELECT id, customer_id, total, status, created_at, expires_at
		 FROM orders WHERE id = $1`, id)

	var o domain.Order
	err := row.Scan(&o.ID, &o.CustomerID, &o.Total, &o.Status, &o.CreatedAt, &o.ExpiresAt)
	if err == sql.ErrNoRows {
		return nil, domain.ErrNotFound
	}
	if err != nil {
		return nil, fmt.Errorf("postgres FindByID: %w", err)
	}
	return &o, nil
}

func (r *PostgresOrderRepo) Save(ctx context.Context, order *domain.Order) error {
	_, err := r.db.ExecContext(ctx,
		`INSERT INTO orders (id, customer_id, total, status, created_at, expires_at)
		 VALUES ($1,$2,$3,$4,$5,$6)
		 ON CONFLICT (id) DO UPDATE SET status=$4, total=$3`,
		order.ID, order.CustomerID, order.Total, order.Status,
		order.CreatedAt, order.ExpiresAt)
	return err
}

func (r *PostgresOrderRepo) FindByCustomerID(ctx context.Context, customerID string) ([]*domain.Order, error) {
	rows, err := r.db.QueryContext(ctx,
		`SELECT id, customer_id, total, status, created_at, expires_at
		 FROM orders WHERE customer_id = $1 ORDER BY created_at DESC`, customerID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var orders []*domain.Order
	for rows.Next() {
		var o domain.Order
		if err := rows.Scan(&o.ID, &o.CustomerID, &o.Total, &o.Status, &o.CreatedAt, &o.ExpiresAt); err != nil {
			return nil, err
		}
		orders = append(orders, &o)
	}
	return orders, rows.Err()
}

func (r *PostgresOrderRepo) FindByStatus(ctx context.Context, status domain.OrderStatus) ([]*domain.Order, error) {
	// similar implementation
	return nil, nil
}

func (r *PostgresOrderRepo) Delete(ctx context.Context, id string) error {
	_, err := r.db.ExecContext(ctx, `DELETE FROM orders WHERE id = $1`, id)
	return err
}
infrastructure/memory_repo.go — In-Memory Implementation (for tests)
package infrastructure

import (
	"context"
	"sync"

	"myapp/domain"
)

// InMemoryOrderRepo — used in unit tests, no database needed
type InMemoryOrderRepo struct {
	mu     sync.RWMutex
	orders map[string]*domain.Order
}

func NewInMemoryOrderRepo() domain.OrderRepository {
	return &InMemoryOrderRepo{
		orders: make(map[string]*domain.Order),
	}
}

func (r *InMemoryOrderRepo) FindByID(_ context.Context, id string) (*domain.Order, error) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	o, ok := r.orders[id]
	if !ok {
		return nil, domain.ErrNotFound
	}
	return o, nil
}

func (r *InMemoryOrderRepo) Save(_ context.Context, order *domain.Order) error {
	r.mu.Lock()
	defer r.mu.Unlock()
	r.orders[order.ID] = order
	return nil
}

func (r *InMemoryOrderRepo) FindByCustomerID(_ context.Context, customerID string) ([]*domain.Order, error) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	var result []*domain.Order
	for _, o := range r.orders {
		if o.CustomerID == customerID {
			result = append(result, o)
		}
	}
	return result, nil
}

func (r *InMemoryOrderRepo) FindByStatus(_ context.Context, status domain.OrderStatus) ([]*domain.Order, error) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	var result []*domain.Order
	for _, o := range r.orders {
		if o.Status == status {
			result = append(result, o)
		}
	}
	return result, nil
}

func (r *InMemoryOrderRepo) Delete(_ context.Context, id string) error {
	r.mu.Lock()
	defer r.mu.Unlock()
	delete(r.orders, id)
	return nil
}
application/order_service.go — Business Logic (no SQL!)
package application

import (
	"context"
	"fmt"
	"time"

	"myapp/domain"
)

type OrderService struct {
	repo domain.OrderRepository // depends on interface, not concrete type
}

func NewOrderService(repo domain.OrderRepository) *OrderService {
	return &OrderService{repo: repo}
}

func (s *OrderService) CancelOrder(ctx context.Context, orderID string) error {
	order, err := s.repo.FindByID(ctx, orderID)
	if err != nil {
		return fmt.Errorf("cancel order: %w", err)
	}

	if !order.CanBeCancelled() {
		return fmt.Errorf("order %s cannot be cancelled", orderID)
	}

	order.Status = domain.StatusCancelled
	return s.repo.Save(ctx, order)
}

func (s *OrderService) GetCustomerOrders(ctx context.Context, customerID string) ([]*domain.Order, error) {
	return s.repo.FindByCustomerID(ctx, customerID)
}
application/order_service_test.go — Test with InMemoryRepo
package application_test

import (
	"context"
	"testing"
	"time"

	"myapp/application"
	"myapp/domain"
	"myapp/infrastructure"
)

func TestCancelOrder_Success(t *testing.T) {
	// Use InMemoryRepo — no database, no setup needed
	repo := infrastructure.NewInMemoryOrderRepo()
	svc := application.NewOrderService(repo)
	ctx := context.Background()

	// Seed test data
	order := &domain.Order{
		ID:         "ORD-001",
		CustomerID: "CUST-001",
		Status:     domain.StatusPending,
		ExpiresAt:  time.Now().Add(24 * time.Hour),
	}
	repo.Save(ctx, order)

	// Act
	err := svc.CancelOrder(ctx, "ORD-001")

	// Assert
	if err != nil {
		t.Fatalf("expected no error, got: %v", err)
	}

	updated, _ := repo.FindByID(ctx, "ORD-001")
	if updated.Status != domain.StatusCancelled {
		t.Errorf("expected cancelled, got %s", updated.Status)
	}
}

func TestCancelOrder_ExpiredOrder(t *testing.T) {
	repo := infrastructure.NewInMemoryOrderRepo()
	svc := application.NewOrderService(repo)
	ctx := context.Background()

	order := &domain.Order{
		ID:        "ORD-002",
		Status:    domain.StatusPending,
		ExpiresAt: time.Now().Add(-1 * time.Hour), // already expired
	}
	repo.Save(ctx, order)

	err := svc.CancelOrder(ctx, "ORD-002")
	if err == nil {
		t.Fatal("expected error for expired order")
	}
}
cmd/main.go — Wiring with Dependency Injection
package main

import (
	"database/sql"
	"log"
	"net/http"

	_ "github.com/lib/pq"
	"myapp/application"
	"myapp/infrastructure"
	"myapp/presentation"
)

func main() {
	db, err := sql.Open("postgres", "postgres://user:pass@localhost/orders")
	if err != nil {
		log.Fatal(err)
	}
	defer db.Close()

	// Dependency Injection — swap repo here, nothing else changes
	repo := infrastructure.NewPostgresOrderRepo(db)
	// repo := infrastructure.NewInMemoryOrderRepo()  // ← swap for tests

	svc := application.NewOrderService(repo)
	handler := presentation.NewOrderHandler(svc)

	http.HandleFunc("/orders/", handler.GetOrder)
	log.Println("Server running on :8080")
	log.Fatal(http.ListenAndServe(":8080", nil))
}

Best Practices

✅
Doควรทำ
  • Define the interface in the domain layer, not infrastructureDefine interface ใน domain layer ไม่ใช่ infrastructure
  • Return domain errors like ErrNotFound, not sql.ErrNoRowsReturn domain errors เช่น ErrNotFound ไม่ใช่ sql.ErrNoRows
  • Use Context on every method to support cancellationใช้ Context ทุก method เพื่อรองรับ cancellation
  • Always use an InMemoryRepo for unit testsInMemoryRepo สำหรับ unit tests ทุกครั้ง
  • One repository per aggregate root1 repository ต่อ 1 aggregate root
⚠️
Cautionระวัง
  • Don't leak SQL errors up into the business layerอย่า leak SQL errors ขึ้นมา business layer
  • Don't put business logic in the repositoryอย่าใส่ business logic ใน repository
  • Don't build a generic repository with no type safetyอย่าสร้าง Generic Repository ที่ไม่มี type safety
  • Don't query across aggregates in one repositoryอย่า query ข้าม aggregate ใน 1 repository
  • Don't forget to Close() rows in query loopsอย่าลืม Close() rows ใน query loops
