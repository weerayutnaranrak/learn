Architecture Pattern #11

Service-Oriented Architecture

ระบบที่ประกอบด้วย Services อิสระที่สื่อสารกันผ่าน Standard Protocol — รากฐานของ Enterprise Integration

SOA คืออะไร?

SOA (Service-Oriented Architecture) คือ architectural style ที่ระบบซอฟต์แวร์ถูกออกแบบในรูปของ Services ที่มีขอบเขตชัดเจน สามารถค้นหาได้ผ่าน Registry และสื่อสารกันผ่าน Standard Protocol เช่น SOAP, REST หรือ Message Queue

🏭
Service Provider
ผู้ให้บริการ — publish service เข้า registry พร้อม contract
  • Implement business logic
  • Expose เป็น endpoint
  • Publish WSDL/OpenAPI
  • Handle versioning
💻
Service Consumer
ผู้ใช้บริการ — ค้นหา service จาก registry แล้วเรียกใช้
  • ค้นหาจาก Registry
  • Bind ตาม contract
  • Handle errors/timeouts
  • Manage retries
📋
Service Registry
ตัวกลาง — เก็บรายการ services ที่ available พร้อม metadata
  • UDDI (Universal Description)
  • Discovery & Integration
  • Health checking
  • Metadata management

Enterprise Service Bus (ESB)

ESB เป็นหัวใจของ SOA — ทำหน้าที่เป็น middleware ที่เชื่อม services เข้าหากัน โดยจัดการ routing, transformation, orchestration, และ protocol mediation

🚌 Enterprise Service Bus
Middleware backbone ที่เชื่อม services ทั้งหมดเข้าหากัน
🔀 Message Routing
🔄 Data Transformation
🎭 Protocol Mediation
🔒 Security & Auth
📊 Monitoring
📝 Audit Logging
⚡ Load Balancing

คำอธิบาย service interface — operations, parameters, data types, endpoints

กำหนด structure ของ request/response — XML Schema, JSON Schema
📜

ข้อตกลง quality of service — availability 99.9%, latency < 200ms, throughput

ตัวอย่าง Code (Go)

registry/registry.go — Service Registry
package registry

import (
	"fmt"
	"sync"
	"time"
)

type ServiceEndpoint struct {
	Name     string
	Version  string
	URL      string
	Protocol string // "soap", "rest", "grpc"
	WSDL     string
	Healthy  bool
	RegisteredAt time.Time
}

type ServiceRegistry struct {
	mu       sync.RWMutex
	services map[string]*ServiceEndpoint
}

func NewRegistry() *ServiceRegistry {
	return &ServiceRegistry{
		services: make(map[string]*ServiceEndpoint),
	}
}

// Publish — Service Provider registers itself
func (r *ServiceRegistry) Publish(svc *ServiceEndpoint) error {
	r.mu.Lock()
	defer r.mu.Unlock()

	key := fmt.Sprintf("%s@%s", svc.Name, svc.Version)
	svc.RegisteredAt = time.Now()
	svc.Healthy = true
	r.services[key] = svc
	return nil
}

// Find — Service Consumer discovers a service
func (r *ServiceRegistry) Find(name, version string) (*ServiceEndpoint, error) {
	r.mu.RLock()
	defer r.mu.RUnlock()

	key := fmt.Sprintf("%s@%s", name, version)
	svc, ok := r.services[key]
	if !ok || !svc.Healthy {
		return nil, fmt.Errorf("service %s v%s not available", name, version)
	}
	return svc, nil
}

esb/esb.go — Enterprise Service Bus (simplified)
package esb

import (
	"context"
	"fmt"
	"log"
	"time"
)

type Message struct {
	ID          string
	ServiceName string
	Version     string
	Operation   string
	Payload     []byte
	Headers     map[string]string
	Timestamp   time.Time
}

type Handler func(ctx context.Context, msg Message) ([]byte, error)

type ESB struct {
	routes      map[string]Handler
	middlewares []func(Handler) Handler
}

func NewESB() *ESB {
	return &ESB{
		routes: make(map[string]Handler),
	}
}

// RegisterService wires a service into the ESB
func (e *ESB) RegisterService(name, operation string, handler Handler) {
	key := fmt.Sprintf("%s.%s", name, operation)
	// Apply middleware chain
	h := handler
	for i := len(e.middlewares) - 1; i >= 0; i-- {
		h = e.middlewares[i](h)
	}
	e.routes[key] = h
}

// Use adds ESB middleware (logging, auth, transform)
func (e *ESB) Use(mw func(Handler) Handler) {
	e.middlewares = append(e.middlewares, mw)
}

// Route dispatches a message to the correct service
func (e *ESB) Route(ctx context.Context, msg Message) ([]byte, error) {
	key := fmt.Sprintf("%s.%s", msg.ServiceName, msg.Operation)
	handler, ok := e.routes[key]
	if !ok {
		return nil, fmt.Errorf("no handler for %s", key)
	}
	return handler(ctx, msg)
}

// LoggingMiddleware — ESB cross-cutting concern
func LoggingMiddleware(next Handler) Handler {
	return func(ctx context.Context, msg Message) ([]byte, error) {
		log.Printf("[ESB] routing %s.%s id=%s", msg.ServiceName, msg.Operation, msg.ID)
		start := time.Now()
		resp, err := next(ctx, msg)
		log.Printf("[ESB] completed in %v err=%v", time.Since(start), err)
		return resp, err
	}
}
cmd/main.go — Wiring it all together
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"

	"myapp/esb"
	"myapp/registry"
	"myapp/services"
)

func main() {
	// 1. Setup Service Registry
	reg := registry.NewRegistry()

	// 2. Setup ESB with middleware
	bus := esb.NewESB()
	bus.Use(esb.LoggingMiddleware)

	// 3. Register Payment Service (Provider publishes itself)
	paymentSvc := services.NewPaymentService("https://payment-gateway.internal")
	reg.Publish(®istry.ServiceEndpoint{
		Name:     "PaymentService",
		Version:  "1.0",
		URL:      "http://localhost:8081",
		Protocol: "rest",
	})

	// 4. Wire service operations to ESB
	bus.RegisterService("PaymentService", "processPayment", func(ctx context.Context, msg esb.Message) ([]byte, error) {
		var req services.PaymentRequest
		if err := json.Unmarshal(msg.Payload, &req); err != nil {
			return nil, err
		}
		resp, err := paymentSvc.ProcessPayment(ctx, req)
		if err != nil {
			return nil, err
		}
		return json.Marshal(resp)
	})

	// 5. Consumer discovers and calls service via ESB
	endpoint, err := reg.Find("PaymentService", "1.0")
	if err != nil {
		log.Fatal("service not found:", err)
	}
	fmt.Printf("Found service at: %s\n", endpoint.URL)

	payload, _ := json.Marshal(services.PaymentRequest{
		OrderID: "ORD-001", Amount: 299.99, Currency: "THB", Method: "credit_card",
	})

	result, err := bus.Route(context.Background(), esb.Message{
		ID:          "MSG-001",
		ServiceName: "PaymentService",
		Operation:   "processPayment",
		Payload:     payload,
	})
	if err != nil {
		log.Fatal(err)
	}

	fmt.Printf("Payment result: %s\n", result)
}

เมื่อไหร่ควรใช้ SOA?

✅
เหมาะกับ
  • Large enterprise ที่มี legacy systems หลายระบบ
  • ต้องการ integrate ระบบที่ใช้ต่าง protocol (SOAP, REST, MQ)
  • Business processes ซับซ้อน ต้อง orchestrate หลาย services
  • ต้องการ governance และ SLA ที่ชัดเจน
  • Enterprise-wide service reuse
⚠️
ระวัง
  • ESB กลายเป็น single point of failure
  • "ESB spaghetti" — logic รั่วเข้า ESB มากเกินไป
  • ค่าใช้จ่าย ESB software สูง (Oracle, MuleSoft, IBM)
  • Performance overhead จาก transformation
  • Complexity สูงสำหรับ small team
