Client-Server Pattern

The foundation of every kind of networked system — the Client sends a Request, the Server processes it and sends a Response back, separating the concerns of requester and providerรากฐานของระบบ Networked ทุกประเภท — Client ส่ง Request, Server ประมวลผลและส่ง Response กลับ ทำให้แยก Concern ระหว่างผู้ขอและผู้ให้บริการ

💡 What is the Client-Server Pattern?Client-Server Pattern คืออะไร?

server/main.go — Go HTTP ServerGo
package main

import (
    "encoding/json"
    "log"
    "net/http"
    "sync"
    "time"
)

// ── Simple In-Memory Product Server ──────────────────────────────────────

type Product struct {
    ID        int       `json:"id"`
    Name      string    `json:"name"`
    Price     float64   `json:"price"`
    CreatedAt time.Time `json:"created_at"`
}

type ProductStore struct {
    mu       sync.RWMutex
    products map[int]*Product
    nextID   int
}

func NewStore() *ProductStore {
    s := &ProductStore{products: make(map[int]*Product), nextID: 1}
    // Seed data
    s.products[1] = &Product{1, "MacBook Pro", 89900, time.Now()}
    s.products[2] = &Product{2, "iPhone 15", 32900, time.Now()}
    s.nextID = 3
    return s
}

type Server struct{ store *ProductStore }

func (s *Server) handleProducts(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "application/json")

    switch r.Method {
    case http.MethodGet:
        s.store.mu.RLock()
        defer s.store.mu.RUnlock()
        products := make([]*Product, 0, len(s.store.products))
        for _, p := range s.store.products { products = append(products, p) }
        json.NewEncoder(w).Encode(products)

    case http.MethodPost:
        var input struct{ Name string; Price float64 }
        if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
            http.Error(w, `{"error":"invalid body"}`, http.StatusBadRequest)
            return
        }
        s.store.mu.Lock()
        defer s.store.mu.Unlock()
        p := &Product{
            ID: s.store.nextID, Name: input.Name,
            Price: input.Price, CreatedAt: time.Now(),
        }
        s.store.products[p.ID] = p
        s.store.nextID++
        w.WriteHeader(http.StatusCreated)
        json.NewEncoder(w).Encode(p)

    default:
        http.Error(w, `{"error":"method not allowed"}`, http.StatusMethodNotAllowed)
    }
}

func (s *Server) handleHealth(w http.ResponseWriter, r *http.Request) {
    json.NewEncoder(w).Encode(map[string]string{
        "status": "ok", "time": time.Now().Format(time.RFC3339),
    })
}

func main() {
    store  := NewStore()
    server := &Server{store: store}

    mux := http.NewServeMux()
    mux.HandleFunc("/products", server.handleProducts)
    mux.HandleFunc("/health",   server.handleHealth)

    // Middleware: logging
    handler := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        mux.ServeHTTP(w, r)
        log.Printf("%s %s %v", r.Method, r.URL.Path, time.Since(start))
    })

    log.Println("Server listening on :8080")
    log.Fatal(http.ListenAndServe(":8080", handler))
}
client/main.go — Go HTTP ClientGo
package main

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "log"
    "net/http"
    "time"
)

// ── ProductClient — wraps HTTP communication ──────────────────────────────

type Product struct {
    ID    int     `json:"id"`
    Name  string  `json:"name"`
    Price float64 `json:"price"`
}

type ProductClient struct {
    baseURL    string
    httpClient *http.Client
}

func NewProductClient(baseURL string) *ProductClient {
    return &ProductClient{
        baseURL: baseURL,
        httpClient: &http.Client{
            Timeout: 10 * time.Second,
            // Transport สำหรับ connection pooling
            Transport: &http.Transport{
                MaxIdleConns:       100,
                MaxIdleConnsPerHost: 10,
                IdleConnTimeout:    90 * time.Second,
            },
        },
    }
}

func (c *ProductClient) ListProducts(ctx context.Context) ([]Product, error) {
    req, err := http.NewRequestWithContext(ctx, http.MethodGet,
        c.baseURL+"/products", nil)
    if err != nil {
        return nil, fmt.Errorf("create request: %w", err)
    }
    req.Header.Set("Accept", "application/json")

    resp, err := c.httpClient.Do(req)
    if err != nil {
        return nil, fmt.Errorf("do request: %w", err)
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("server returned %d", resp.StatusCode)
    }

    var products []Product
    if err := json.NewDecoder(resp.Body).Decode(&products); err != nil {
        return nil, fmt.Errorf("decode response: %w", err)
    }
    return products, nil
}

func (c *ProductClient) CreateProduct(ctx context.Context, name string, price float64) (*Product, error) {
    body, _ := json.Marshal(map[string]interface{}{"name": name, "price": price})

    req, _ := http.NewRequestWithContext(ctx, http.MethodPost,
        c.baseURL+"/products", bytes.NewReader(body))
    req.Header.Set("Content-Type", "application/json")

    resp, err := c.httpClient.Do(req)
    if err != nil {
        return nil, fmt.Errorf("do request: %w", err)
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusCreated {
        return nil, fmt.Errorf("server returned %d", resp.StatusCode)
    }

    var p Product
    if err := json.NewDecoder(resp.Body).Decode(&p); err != nil {
        return nil, fmt.Errorf("decode: %w", err)
    }
    return &p, nil
}

func main() {
    client := NewProductClient("http://localhost:8080")
    ctx    := context.Background()

    // LIST products
    products, err := client.ListProducts(ctx)
    if err != nil {
        log.Fatalf("list: %v", err)
    }
    fmt.Println("=== สินค้าทั้งหมด ===")
    for _, p := range products {
        fmt.Printf("  [%d] %s — ฿%.2f\n", p.ID, p.Name, p.Price)
    }

    // CREATE product
    newProduct, err := client.CreateProduct(ctx, "AirPods Pro", 9900)
    if err != nil {
        log.Fatalf("create: %v", err)
    }
    fmt.Printf("\n✅ สร้างสินค้าใหม่: [%d] %s\n", newProduct.ID, newProduct.Name)
}
websocket/realtime.go — WebSocket Client-Server (Real-time)Go
package main

import (
    "encoding/json"
    "log"
    "net/http"
    "sync"
    "golang.org/x/net/websocket"
)

// ── WebSocket: Bi-directional persistent connection ────────────────────────

type Message struct {
    Type    string      `json:"type"`
    Payload interface{} `json:"payload"`
}

type Hub struct {
    mu      sync.RWMutex
    clients map[*websocket.Conn]bool
}

func (h *Hub) broadcast(msg Message) {
    data, _ := json.Marshal(msg)
    h.mu.RLock()
    defer h.mu.RUnlock()
    for conn := range h.clients {
        websocket.Message.Send(conn, string(data))
    }
}

func (h *Hub) handleWS(ws *websocket.Conn) {
    // Register
    h.mu.Lock()
    h.clients[ws] = true
    h.mu.Unlock()
    log.Printf("Client connected: %s", ws.RemoteAddr())

    // Send welcome
    h.broadcast(Message{Type: "connected", Payload: "Welcome!"})

    // Read loop (Client → Server)
    for {
        var raw string
        if err := websocket.Message.Receive(ws, &raw); err != nil {
            break
        }
        var msg Message
        json.Unmarshal([]byte(raw), &msg)

        // Echo back to all clients (broadcast)
        h.broadcast(Message{Type: "chat", Payload: msg.Payload})
    }

    // Deregister
    h.mu.Lock()
    delete(h.clients, ws)
    h.mu.Unlock()
    log.Printf("Client disconnected: %s", ws.RemoteAddr())
}

func main() {
    hub := &Hub{clients: make(map[*websocket.Conn]bool)}

    http.Handle("/ws", websocket.Handler(hub.handleWS))
    http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
        w.Write([]byte(`<html><body>
          <script>
            const ws = new WebSocket('ws://localhost:8080/ws');
            ws.onmessage = e => console.log(JSON.parse(e.data));
            ws.onopen = () => ws.send(JSON.stringify({type:'chat', payload:'Hello!'}));
          </script>
          </body></html>`))
    })

    log.Println("WebSocket server on :8080")
    log.Fatal(http.ListenAndServe(":8080", nil))
}
