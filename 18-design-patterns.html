🏭
Factory Method
Creational — Define an interface for creating an object, let subclasses decide which class to instantiate
ปัญหา: code ต้องสร้าง object แต่ไม่อยากผูกกับ concrete type — อยากเปลี่ยน implementation ได้ทีหลัง
วิธีแก้: ซ่อนการสร้าง object ไว้ใน factory function/method — caller รู้แค่ interface ไม่รู้ concrete type
❌ สร้าง concrete type โดยตรง ผิดหลักการ 
// caller รู้ว่าเป็น SMTPMailer
// ถ้าอยากเปลี่ยนเป็น SendGrid ต้องแก้ทุกจุด
func sendWelcome(user User) {
    mailer := &SMTPMailer{
        host: "smtp.co.th",
        port: 587,
    }
    mailer.Send(user.Email, "Welcome!")

✅ Factory ซ่อน concrete type ถูกหลักการ 
type Mailer interface {
    Send(to, body string) error
}

// Factory function — สร้างผ่านนี้เสมอ
func NewMailer(cfg Config) Mailer {
    switch cfg.Provider {
    case "sendgrid":
        return &SendGridMailer{apiKey: cfg.Key}
    case "ses":
        return &SESMailer{region: cfg.Region}
    default:
        return &SMTPMailer{host: cfg.Host}
    }
}

// caller ไม่รู้ concrete type
func sendWelcome(m Mailer, user User) {
    m.Send(user.Email, "Welcome!")

// main.go — เปลี่ยน provider ที่จุดเดียว

🔧
Builder
Creational — Construct complex objects step by step
ปัญหา: struct ที่มี field เยอะและบาง field เป็น optional — constructor argument ยาวมากจนอ่านไม่ออกว่า argument ไหนคืออะไร
วิธีแก้: สร้าง Builder type ที่มี method ชื่อชัดเจน chain กันได้ แล้วเรียก .Build() ตอนท้าย
❌ Constructor ที่อ่านไม่ออก ผิดหลักการ

✅ Builder — chain method อ่านออกทันที ถูกหลักการ 
type ServerBuilder struct{ s Server }

func NewServerBuilder(host string, port int) *ServerBuilder {
    return &ServerBuilder{s: Server{Host: host, Port: port}}
}

func (b *ServerBuilder) WithTLS() *ServerBuilder {
    b.s.TLS = true; return b
}
func (b *ServerBuilder) WithDebug() *ServerBuilder {
    b.s.Debug = true; return b
}
func (b *ServerBuilder) WithTimeout(t int) *ServerBuilder {
    b.s.Timeout = t; return b
}
func (b *ServerBuilder) WithBasePath(p string) *ServerBuilder {
    b.s.BasePath = p; return b
}
func (b *ServerBuilder) Build() (*Server, error) {
    if b.s.Host == "" {
        return nil, errors.New("host required")
    }
    return &b.s, nil
}

// อ่านออกทันทีว่าทำอะไร

class ServerBuilder {
  private config: Partial = {};

  constructor(host: string, port: number) {
    this.config.host = host;
    this.config.port = port;
  }

  withTLS(): this {
    this.config.tls = true; return this;
  }
  withDebug(): this {
    this.config.debug = true; return this;
  }
  withTimeout(t: number): this {
    this.config.timeout = t; return this;
  }
  withBasePath(p: string): this {
    this.config.basePath = p; return this;
  }

  build(): ServerConfig {
    if (!this.config.host) throw new Error('host required');
    return this.config as ServerConfig;
  }
}

// อ่านออกทันทีว่าทำอะไร
const srv = new ServerBuilder('localhost', 8080)
  .withTLS()
  .withTimeout(30)
  .withBasePath('/api')
  .build();
Functional Options (Go idiom): อีกแบบที่นิยมใน Go คือ func(o *Options) option pattern — NewServer(host, port, WithTLS(), WithTimeout(30)) — ให้ผลเหมือนกันแต่ไม่ต้องมี Builder type

Decorator
Structural — Attach additional behavior to objects by wrapping them
ปัญหา: อยากเพิ่ม behavior เช่น logging, caching, retry โดยไม่แก้ original struct — และอยาก wrap ซ้อนกันได้หลายชั้น
วิธีแก้: สร้าง wrapper type ที่ implement interface เดิม แต่ใส่ logic เพิ่มก่อน/หลังแล้ว delegate ไปยัง inner
✅ Decorator — เพิ่ม logging + caching โดยไม่แก้ original ถูกหลักการ 
type OrderRepository interface {
    FindByID(id string) (*Order, error)
    Save(o *Order) error
}

// Original implementation
type PostgresOrderRepo struct{ db *sql.DB }
func (r *PostgresOrderRepo) FindByID(id string) (*Order, error) { /* DB query */ }
func (r *PostgresOrderRepo) Save(o *Order) error { /* DB write */ }

// Decorator 1: Logging — wrap ซ้อนบน original
type LoggingOrderRepo struct {
    inner  OrderRepository
    logger *slog.Logger
}

func (r *LoggingOrderRepo) FindByID(id string) (*Order, error) {
    start := time.Now()
    order, err := r.inner.FindByID(id) // delegate
    r.logger.Info("FindByID", "id", id, "ms", time.Since(start).Milliseconds(), "err", err)
    return order, err
}
func (r *LoggingOrderRepo) Save(o *Order) error { return r.inner.Save(o) }

// Decorator 2: Caching — wrap ซ้อนบน logging
type CachingOrderRepo struct {
    inner OrderRepository
    cache map[string]*Order
}

func (r *CachingOrderRepo) FindByID(id string) (*Order, error) {
    if o, ok := r.cache[id]; ok {
        return o, nil // cache hit
    }
    o, err := r.inner.FindByID(id) // miss → delegate
    if err == nil { r.cache[id] = o }
    return o, err
}
func (r *CachingOrderRepo) Save(o *Order) error {
    delete(r.cache, o.ID) // invalidate
    return r.inner.Save(o)
}

// Wire-up: ซ้อน decorator หลายชั้น
repo := &PostgresOrderRepo{db: db}
repo = &LoggingOrderRepo{inner: repo, logger: logger}
repo = &CachingOrderRepo{inner: repo, cache: map[string]*Order{}}

// caller ไม่รู้เลยว่ามีกี่ชั้น — ใช้แค่ interface
svc := NewOrderService(repo)
interface OrderRepository {
  findById(id: string): Promise;
  save(order: Order): Promise;
}

// Original implementation
class PostgresOrderRepo implements OrderRepository {
  constructor(private db: Pool) {}
  async findById(id: string) { /* DB query */ return null; }
  async save(o: Order) { /* DB write */ }
}

// Decorator 1: Logging — wrap ซ้อนบน original
class LoggingOrderRepo implements OrderRepository {
  constructor(
    private inner: OrderRepository,
    private logger: Logger,
  ) {}

  async findById(id: string) {
    const start = Date.now();
    const order = await this.inner.findById(id); // delegate
    this.logger.info('findById', { id, ms: Date.now() - start });
    return order;
  }
  async save(o: Order) { return this.inner.save(o); }
}

// Decorator 2: Caching — wrap ซ้อนบน logging
class CachingOrderRepo implements OrderRepository {
  private cache = new Map();

  constructor(private inner: OrderRepository) {}

  async findById(id: string) {
    if (this.cache.has(id)) return this.cache.get(id)!; // cache hit
    const order = await this.inner.findById(id); // miss → delegate
    if (order) this.cache.set(id, order);
    return order;
  }
  async save(o: Order) {
    this.cache.delete(o.id); // invalidate
    return this.inner.save(o);
  }
}

// Wire-up: ซ้อน decorator หลายชั้น
let repo: OrderRepository = new PostgresOrderRepo(db);
repo = new LoggingOrderRepo(repo, logger);
repo = new CachingOrderRepo(repo);

// caller ไม่รู้เลยว่ามีกี่ชั้น
const svc = new OrderService(repo);
Middleware = Decorator: HTTP middleware ทุกตัวที่เขียนใน Go (func(http.Handler) http.Handler) คือ Decorator pattern — logging, auth, rate-limit ล้วนซ้อนกันบน handler จริง

Strategy
Behavioral — Define a family of algorithms, encapsulate each one, make them interchangeable
ปัญหา: behavior หลายแบบที่เลือกได้ตอน runtime — ถ้าใช้ if/switch ทุกครั้งที่เพิ่มแบบใหม่ต้องแก้ code เดิม
วิธีแก้: ห่อแต่ละ algorithm ใน struct ของตัวเอง inject เข้ามาทาง interface — เปลี่ยนได้ตอน runtime
✅ Strategy — เปลี่ยน sorting algorithm โดยไม่แก้ caller ถูกหลักการ 
// Strategy interface
type SortStrategy interface {
    Sort(data []int) []int
}

// Concrete strategies
type BubbleSort  struct{}
type QuickSort   struct{}
type MergeSort   struct{}

func (BubbleSort) Sort(d []int) []int  { /* bubble sort */; return d }
func (QuickSort)  Sort(d []int) []int  { /* quick sort  */; return d }
func (MergeSort)  Sort(d []int) []int  { /* merge sort  */; return d }

// Context — ใช้ strategy ที่ inject มา
type DataProcessor struct {
    strategy SortStrategy
}

func (p *DataProcessor) SetStrategy(s SortStrategy) {
    p.strategy = s
}

func (p *DataProcessor) Process(data []int) []int {
    return p.strategy.Sort(data)
}

// เปลี่ยน strategy ตอน runtime
proc := &DataProcessor{strategy: QuickSort{}}
result := proc.Process(bigData) // quick sort

proc.SetStrategy(MergeSort{}) // swap strategy
result = proc.Process(bigData) // merge sort

// Concrete strategies
class BubbleSort implements SortStrategy {

👁️
Observer
Behavioral — Define a one-to-many dependency so when one object changes state, all dependents are notified
ปัญหา: เมื่อ object หนึ่งเปลี่ยน state อยากให้ object อื่นๆ รู้โดยอัตโนมัติ แต่ไม่อยาก coupling กันโดยตรง
วิธีแก้: ให้ Subject เก็บ list ของ Observer — เมื่อ state เปลี่ยนเรียก Notify() ให้ทุกตัวโดยที่ Subject ไม่รู้จักรายละเอียดของ Observer
✅ Observer — Order state เปลี่ยน → แจ้งหลายระบบอัตโนมัติ ถูกหลักการ 
// Observer interface
type OrderObserver interface {
    OnOrderPlaced(o Order)
}

// Subject: เก็บ list of observers
type OrderService struct {
    observers []OrderObserver
    repo      OrderRepository
}

func (s *OrderService) Subscribe(obs OrderObserver) {
    s.observers = append(s.observers, obs)
}

func (s *OrderService) notify(o Order) {
    for _, obs := range s.observers {
        go obs.OnOrderPlaced(o) // async notify
    }
}

func (s *OrderService) PlaceOrder(o Order) error {
    if err := s.repo.Save(o); err != nil {
        return err
    }
    s.notify(o) // broadcast — ไม่รู้ว่ามีใคร subscribe
    return nil
}

// Observers: ต่างคนต่างรับแล้วทำของตัวเอง
type EmailNotifier struct{ mailer Mailer }
func (e *EmailNotifier) OnOrderPlaced(o Order) {
    e.mailer.Send(o.Email, "Order "+o.ID+" placed!")
}

type InventoryService struct{ repo InventoryRepo }
func (i *InventoryService) OnOrderPlaced(o Order) {
    i.repo.Reserve(o.Items)
}

type AuditLogger struct{ log *slog.Logger }
func (a *AuditLogger) OnOrderPlaced(o Order) {
    a.log.Info("order_placed", "id", o.ID, "total", o.Total)

// Observer interface
interface OrderObserver {
  onOrderPlaced(order: Order): void;
}

// Subject: เก็บ list of observers
class OrderService {
  private observers: OrderObserver[] = [];

  constructor(private repo: OrderRepository) {}

  subscribe(obs: OrderObserver) {
    this.observers.push(obs);
  }

  private notify(order: Order) {
    for (const obs of this.observers) {
      obs.onOrderPlaced(order); // each handles async internally
    }
  }

  async placeOrder(order: Order): Promise {
    await this.repo.save(order);
    this.notify(order); // broadcast — ไม่รู้ว่ามีใคร subscribe
  }
}

// Observers: ต่างคนต่างรับแล้วทำของตัวเอง
class EmailNotifier implements OrderObserver {
  constructor(private mailer: Mailer) {}
  onOrderPlaced(order: Order) {
    this.mailer.send(order.email, `Order ${order.id} placed!`);
  }
}

class InventoryService implements OrderObserver {
  constructor(private repo: InventoryRepo) {}
  onOrderPlaced(order: Order) { this.repo.reserve(order.items); }
}

class AuditLogger implements OrderObserver {
  constructor(private logger: Logger) {}
  onOrderPlaced(order: Order) {
    this.logger.info('order_placed', { id: order.id, total: order.total });
  }
}

// Wire-up
const svc = new OrderService(repo);
svc.subscribe(new EmailNotifier(mailer));
svc.subscribe(new InventoryService(invRepo));
svc.subscribe(new AuditLogger(logger));
Observer vs Event-Driven: Observer เป็น in-process (function call), Event-Driven ข้าม process ผ่าน message broker — ใช้ Observer ก่อนเมื่อยังอยู่ใน service เดียวกัน
🔗
Chain of Responsibility
Behavioral — Pass requests along a chain of handlers
ปัญหา: request ต้องผ่านขั้นตอนหลายอย่าง (auth → rate-limit → validate → handle) แต่ไม่อยากให้ handler รู้จักกันโดยตรง
วิธีแก้: เชื่อม handler เป็น chain — แต่ละตัวจัดการส่วนของตัวแล้วเรียก next หรือหยุดถ้า error
✅ HTTP Middleware chain — Go idiom ถูกหลักการ 
// Middleware type ใน Go คือ func(http.Handler) http.Handler
type Middleware func(http.Handler) http.Handler

// Middleware 1: logging
func Logging(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        next.ServeHTTP(w, r) // ส่งต่อไป
        log.Printf("%s %s %v", r.Method, r.URL.Path, time.Since(start))
    })
}

// Middleware 2: auth
func Auth(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        token := r.Header.Get("Authorization")
        if !isValid(token) {
            http.Error(w, "Unauthorized", 401)
            return // หยุด chain ที่นี่
        }
        next.ServeHTTP(w, r)
    })
}

// Middleware 3: rate limit
func RateLimit(rps int) Middleware {
    limiter := rate.NewLimiter(rate.Limit(rps), rps)
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            if !limiter.Allow() {
                http.Error(w, "Too Many Requests", 429)
                return
            }
            next.ServeHTTP(w, r)
        })
    }
}

// Chain helper
func Chain(h http.Handler, mw ...Middleware) http.Handler {
    for i := len(mw) - 1; i >= 0; i-- {
        h = mw[i](h) // wrap จากในออกนอก
    }
    return h
}

// Wire-up: อ่านออกว่า request ผ่านอะไรบ้าง
handler := Chain(
    orderHandler,
    Logging,
    Auth,
    RateLimit(100),
)
// Middleware type ใน Express/Node
type Middleware = (req: Request, res: Response, next: NextFunction) => void;

// Middleware 1: logging
const logging: Middleware = (req, res, next) => {
  const start = Date.now();
  res.on('finish', () => {
    console.log(`${req.method} ${req.path} ${res.statusCode} ${Date.now()-start}ms`);
  });
  next(); // ส่งต่อไป
};

// Middleware 2: auth
const auth: Middleware = (req, res, next) => {
  const token = req.headers.authorization;
  if (!isValid(token)) {
    res.status(401).json({ error: 'Unauthorized' });
    return; // หยุด chain ที่นี่
  }
  next();
};

// Middleware 3: rate limit
function rateLimit(rps: number): Middleware {
  const limiter = new RateLimiter(rps);
  return (req, res, next) => {
    if (!limiter.allow()) {
      res.status(429).json({ error: 'Too Many Requests' });
      return;
    }
    next();
  };
}

// Wire-up: อ่านออกว่า request ผ่านอะไรบ้าง
app.use(logging);
app.use(auth);
app.use(rateLimit(100));
app.post('/orders', orderHandler);
🔗 Pattern กับ SOLID
แต่ละ pattern คือ SOLID principle ในทางปฏิบัติ
🏭 Factory Method
D — caller depend on interface ไม่ใช่ concrete
O — เพิ่ม implementation ใหม่ไม่แก้ caller
🔧 Builder
S — แยก construction logic ออกจาก struct
อ่าน code ง่ายขึ้น ลด error จาก positional args
🎀 Decorator
O — เพิ่ม behavior โดยไม่แก้ original
S — แต่ละ decorator มี 1 responsibility
♟️ Strategy
O — เพิ่ม algorithm ใหม่ไม่แก้ context
D — context depend on strategy interface
👁️ Observer
O — เพิ่ม observer ใหม่ไม่แก้ subject
S — subject ไม่รับผิดชอบ logic ของ observer
🔗 Chain of Resp.
S — แต่ละ handler ทำงานเดียว
O — เพิ่ม middleware ใหม่ไม่แก้ handler เดิม
