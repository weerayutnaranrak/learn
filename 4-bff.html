BFF Architecture

Backend for Frontend — build a backend layer tailored to each client type instead of one generic API for everyoneBackend for Frontend — สร้าง Backend Layer ที่ tailored สำหรับแต่ละ Client Type แทนที่จะมี Generic API เดียวสำหรับทุกคน

😣 Problems BFF solvesปัญหาที่ BFF แก้
Why isn't a single generic API enough?ทำไม Generic API เดียวไม่พอ?

When a system has many clients (Web, Mobile App, Smart TV, IoT), each has very different needs:เมื่อระบบมีหลาย client (Web, Mobile App, Smart TV, IoT) แต่ละตัวมีความต้องการต่างกันมาก:

Problem 1ปัญหา 1

Over-fetching / Under-fetching

Web needs lots of data, mobile needs only some — a generic API gives both the same, so mobile downloads more than it needsWeb ต้องการข้อมูลเยอะ, Mobile ต้องการแค่บางส่วน — Generic API ให้มาเท่ากัน ทำให้ mobile โหลดข้อมูลเกินความจำเป็น

Problem 2ปัญหา 2

N+1 Client Calls

Mobile has to call 5 endpoints to render one screen — slow and battery-draining — because the microservices don't know the client contextMobile ต้อง call 5 endpoints เพื่อแสดง 1 หน้า ทำให้ช้า และ battery drain — เพราะ microservices ไม่รู้จัก client context

Problem 3ปัญหา 3

Coupling between client and microservicesCoupling ระหว่าง Client และ Microservices

The client knows every microservice — change the internal structure and the client has to update tooClient รู้จัก microservices ทุกตัว — เปลี่ยน internal structure ทีไร client ต้องอัพเดทตาม

The BFF pattern solves this by: building a separate "backend" for each client type — Web BFF, Mobile BFF, Smart TV BFF — each aggregating data from the microservices and returning only what that client needsBFF Pattern แก้ด้วย: สร้าง "Backend" แยกสำหรับแต่ละ client type — Web BFF, Mobile BFF, Smart TV BFF แต่ละตัว aggregate data จาก microservices แล้วส่งเฉพาะที่ client นั้นต้องการ

clients/upstream.go — Upstream Service Clients (Shared) Go 
package clients

import (
    "context"
    "encoding/json"
    "fmt"
    "net/http"
    "time"
)

// ── Models จาก Upstream Microservices ────────────────────────────────────────

// Product — full model จาก Product Service
type Product struct {
    ID            string    `json:"id"`
    Name          string    `json:"name"`
    Description   string    `json:"description"`    // ยาวมาก
    Price         float64   `json:"price"`
    OriginalPrice float64   `json:"original_price"`
    Currency      string    `json:"currency"`
    Images        []string  `json:"images"`         // หลายรูป
    Category      string    `json:"category"`
    Tags          []string  `json:"tags"`
    Rating        float64   `json:"rating"`
    ReviewCount   int       `json:"review_count"`
    StockCount    int       `json:"stock_count"`
    Weight        float64   `json:"weight_kg"`
    Dimensions    Dimension `json:"dimensions"`
    CreatedAt     time.Time `json:"created_at"`
    UpdatedAt     time.Time `json:"updated_at"`
}

type Dimension struct{ Width, Height, Depth float64 }

// Cart — จาก Cart Service
type Cart struct {
    ID    string      `json:"id"`
    Items []CartItem  `json:"items"`
    Total float64     `json:"total"`
}
type CartItem struct{ ProductID, Name string; Price float64; Qty int }

// Recommendation — จาก Recommendation Service
type Recommendation struct{ ProductID, Name string; Score float64 }

// ── HTTP Clients ───────────────────────────────────────────────────────────────

type ProductClient struct{ baseURL string; http *http.Client }

func NewProductClient(baseURL string) *ProductClient {
    return &ProductClient{
        baseURL: baseURL,
        http:    &http.Client{Timeout: 5 * time.Second},
    }
}

func (c *ProductClient) GetProduct(ctx context.Context, id string) (*Product, error) {
    req, _ := http.NewRequestWithContext(ctx, "GET",
        fmt.Sprintf("%s/products/%s", c.baseURL, id), nil)
    resp, err := c.http.Do(req)
    if err != nil {
        return nil, fmt.Errorf("fetch product: %w", err)
    }
    defer resp.Body.Close()
    var p Product
    return &p, json.NewDecoder(resp.Body).Decode(&p)
}

func (c *ProductClient) GetRecommendations(ctx context.Context, productID string) ([]Recommendation, error) {
    req, _ := http.NewRequestWithContext(ctx, "GET",
        fmt.Sprintf("%s/products/%s/recommendations", c.baseURL, productID), nil)
    resp, err := c.http.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    var recs []Recommendation
    return recs, json.NewDecoder(resp.Body).Decode(&recs)
}

type CartClient struct{ baseURL string; http *http.Client }
func NewCartClient(baseURL string) *CartClient {
    return &CartClient{baseURL: baseURL, http: &http.Client{Timeout: 3 * time.Second}}
}
func (c *CartClient) GetCart(ctx context.Context, userID string) (*Cart, error) {
    req, _ := http.NewRequestWithContext(ctx, "GET",
        fmt.Sprintf("%s/carts/%s", c.baseURL, userID), nil)
    resp, err := c.http.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    var cart Cart
    return &cart, json.NewDecoder(resp.Body).Decode(&cart)
}
webbff/handler.go — Web BFF (Full Data + Aggregation) Go 
package webbff

import (
    "context"
    "encoding/json"
    "net/http"
    "sync"
    "myapp/clients"
)

// ── Web Response Models (tailored for Web) ───────────────────────────────────

// WebProductPage — response สำหรับ Web หน้า Product Detail
// รวมข้อมูลจากหลาย services ในการ call เดียว
type WebProductPage struct {
    Product         WebProduct         `json:"product"`
    Cart            *WebCart           `json:"cart,omitempty"`
    Recommendations []WebRecommendation `json:"recommendations"`
    RelatedProducts []WebProductSummary `json:"related_products"`
}

// WebProduct — ข้อมูลสำหรับ Web (full detail)
type WebProduct struct {
    ID            string   `json:"id"`
    Name          string   `json:"name"`
    Description   string   `json:"description"`   // Web แสดง description เต็ม
    Price         float64  `json:"price"`
    OriginalPrice float64  `json:"original_price"`
    DiscountPct   int      `json:"discount_pct"`  // คำนวณใน BFF
    Currency      string   `json:"currency"`
    Images        []string `json:"images"`         // ทุกรูป
    PrimaryImage  string   `json:"primary_image"`
    Rating        float64  `json:"rating"`
    ReviewCount   int      `json:"review_count"`
    InStock       bool     `json:"in_stock"`
    Tags          []string `json:"tags"`
}

type WebCart struct {
    ItemCount int     `json:"item_count"`
    Total     float64 `json:"total"`
}

type WebRecommendation struct {
    ProductID string  `json:"product_id"`
    Name      string  `json:"name"`
    Score     float64 `json:"score"`
}

type WebProductSummary struct {
    ID    string  `json:"id"`
    Name  string  `json:"name"`
    Price float64 `json:"price"`
    Image string  `json:"image"`
}

// ── Web BFF Handler ───────────────────────────────────────────────────────────

type WebBFFHandler struct {
    products *clients.ProductClient
    carts    *clients.CartClient
}

func NewWebBFFHandler(products *clients.ProductClient, carts *clients.CartClient) *WebBFFHandler {
    return &WebBFFHandler{products: products, carts: carts}
}

// GetProductPage — Aggregate หลาย calls เป็น response เดียวสำหรับ Web
// Fan-out parallel calls แล้วรวมผล
func (h *WebBFFHandler) GetProductPage(w http.ResponseWriter, r *http.Request) {
    productID := r.URL.Query().Get("id")
    userID    := r.URL.Query().Get("user")

    ctx := r.Context()

    // ── Fan-out: call 3 services พร้อมกัน ────────────────────────────────
    var (
        product *clients.Product
        cart    *clients.Cart
        recs    []clients.Recommendation
        mu      sync.Mutex
        wg      sync.WaitGroup
    )

    wg.Add(3)

    go func() {
        defer wg.Done()
        p, err := h.products.GetProduct(ctx, productID)
        if err == nil {
            mu.Lock()
            product = p
            mu.Unlock()
        }
    }()

    go func() {
        defer wg.Done()
        if userID != "" {
            c, err := h.carts.GetCart(ctx, userID)
            if err == nil {
                mu.Lock()
                cart = c
                mu.Unlock()
            }
        }
    }()

    go func() {
        defer wg.Done()
        r, err := h.products.GetRecommendations(ctx, productID)
        if err == nil {
            mu.Lock()
            recs = r
            mu.Unlock()
        }
    }()

    wg.Wait()

    if product == nil {
        http.Error(w, "product not found", http.StatusNotFound)
        return
    }

    // ── Transform: แปลง upstream models เป็น Web models ─────────────────
    webProduct := toWebProduct(product)

    var webCart *WebCart
    if cart != nil {
        webCart = &WebCart{ItemCount: len(cart.Items), Total: cart.Total}
    }

    webRecs := make([]WebRecommendation, len(recs))
    for i, r := range recs {
        webRecs[i] = WebRecommendation{ProductID: r.ProductID, Name: r.Name, Score: r.Score}
    }

    page := WebProductPage{
        Product:         webProduct,
        Cart:            webCart,
        Recommendations: webRecs,
    }

    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(page)
}

func toWebProduct(p *clients.Product) WebProduct {
    primaryImage := ""
    if len(p.Images) > 0 {
        primaryImage = p.Images[0]
    }

    discountPct := 0
    if p.OriginalPrice > p.Price && p.OriginalPrice > 0 {
        discountPct = int((p.OriginalPrice - p.Price) / p.OriginalPrice * 100)
    }

    return WebProduct{
        ID: p.ID, Name: p.Name, Description: p.Description,
        Price: p.Price, OriginalPrice: p.OriginalPrice, DiscountPct: discountPct,
        Currency: p.Currency, Images: p.Images, PrimaryImage: primaryImage,
        Rating: p.Rating, ReviewCount: p.ReviewCount,
        InStock: p.StockCount > 0, Tags: p.Tags,
    }
}
mobilebff/handler.go — Mobile BFF (Lightweight + Bandwidth-conscious) Go 
package mobilebff

import (
    "encoding/json"
    "net/http"
    "myapp/clients"
)

// ── Mobile Response Models (Minimal) ────────────────────────────────────────
// Mobile ใช้ bandwidth จำกัด + CPU น้อยกว่า Web
// ดังนั้น response ต้องเล็กและ simple

// MobileProduct — ข้อมูลสั้นๆ สำหรับ Mobile List View
type MobileProduct struct {
    ID      string  `json:"id"`
    Name    string  `json:"name"`
    Price   float64 `json:"price"`
    Image   string  `json:"image"`    // รูปเดียว ไม่ใช่ array
    Rating  float64 `json:"rating"`
    InStock bool    `json:"in_stock"`
    // ไม่มี: description, dimensions, weight, tags, reviews, etc.
}

// MobileProductDetail — สำหรับ Detail View
type MobileProductDetail struct {
    ID          string  `json:"id"`
    Name        string  `json:"name"`
    Price       float64 `json:"price"`
    Currency    string  `json:"currency"`
    Image       string  `json:"image"`
    Images      []string `json:"images,omitempty"` // max 3 รูป เท่านั้น
    Rating      float64 `json:"rating"`
    ReviewCount int     `json:"review_count"`
    InStock     bool    `json:"in_stock"`
    ShortDesc   string  `json:"short_description"` // ตัดสั้น ไม่เกิน 150 ตัวอักษร
}

// ── Mobile BFF Handler ────────────────────────────────────────────────────────

type MobileBFFHandler struct {
    products *clients.ProductClient
}

func NewMobileBFFHandler(products *clients.ProductClient) *MobileBFFHandler {
    return &MobileBFFHandler{products: products}
}

// GetProduct — Simple single call, minimal response
func (h *MobileBFFHandler) GetProduct(w http.ResponseWriter, r *http.Request) {
    productID := r.URL.Query().Get("id")

    product, err := h.products.GetProduct(r.Context(), productID)
    if err != nil {
        http.Error(w, "not found", http.StatusNotFound)
        return
    }

    // Transform เป็น minimal model
    mobile := toMobileDetail(product)

    w.Header().Set("Content-Type", "application/json")
    // Gzip compress สำหรับ mobile bandwidth
    w.Header().Set("Content-Encoding", "gzip")
    json.NewEncoder(w).Encode(mobile)
}

// GetProductList — Optimized list สำหรับ mobile infinite scroll
func (h *MobileBFFHandler) GetProductList(w http.ResponseWriter, r *http.Request) {
    // mobile ได้รับ cursor-based pagination แทน offset
    // เพราะ infinite scroll ต้องการ cursor
    cursor := r.URL.Query().Get("cursor")
    _ = cursor

    // Call upstream — ใช้ fields filter เพื่อ bandwidth
    // ใน production อาจส่ง fields=id,name,price,image ไปให้ upstream filter
    // ...

    // Return compact list
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(map[string]interface{}{
        "items":      []MobileProduct{},
        "next_cursor": "xyz123",
        "has_more":    false,
    })
}

func toMobileDetail(p *clients.Product) MobileProductDetail {
    primaryImage := ""
    images := []string{}
    for i, img := range p.Images {
        if i == 0 {
            primaryImage = img
        }
        if i < 3 { // Mobile: max 3 รูป
            images = append(images, img)
        }
    }

    // ตัด description ให้สั้น (150 chars)
    shortDesc := p.Description
    if len(shortDesc) > 150 {
        shortDesc = shortDesc[:147] + "..."
    }

    return MobileProductDetail{
        ID:          p.ID,
        Name:        p.Name,
        Price:       p.Price,
        Currency:    p.Currency,
        Image:       primaryImage,
        Images:      images,
        Rating:      p.Rating,
        ReviewCount: p.ReviewCount,
        InStock:     p.StockCount > 0,
        ShortDesc:   shortDesc,
    }
}
cmd/main.go — Run both BFF serversRun ทั้ง 2 BFF Servers Go 
package main

import (
    "log"
    "net/http"
    "myapp/clients"
    "myapp/webbff"
    "myapp/mobilebff"
)

func main() {
    // Shared upstream clients
    productClient := clients.NewProductClient("http://product-service:3001")
    cartClient    := clients.NewCartClient("http://cart-service:3002")

    // ── Web BFF — Port 8081 ──────────────────────────────────────────────
    webHandler := webbff.NewWebBFFHandler(productClient, cartClient)
    webMux := http.NewServeMux()
    webMux.HandleFunc("/api/product-page", webHandler.GetProductPage)

    go func() {
        log.Println("Web BFF running on :8081")
        log.Fatal(http.ListenAndServe(":8081", webMux))
    }()

    // ── Mobile BFF — Port 8082 ───────────────────────────────────────────
    mobileHandler := mobilebff.NewMobileBFFHandler(productClient)
    mobileMux := http.NewServeMux()
    mobileMux.HandleFunc("/api/products",  mobileHandler.GetProductList)
    mobileMux.HandleFunc("/api/product",   mobileHandler.GetProduct)

    log.Println("Mobile BFF running on :8082")
    log.Fatal(http.ListenAndServe(":8082", mobileMux))
}

✅ Summary & Best Practicesสรุปและ Best Practices
Prosข้อดี

Use it whenควรใช้เมื่อ

• You have several client types (Web, Mobile, TV, IoT)
• Clients need to aggregate many APIs
• You want to optimize bandwidth for mobile
• Separate teams (frontend team owns the BFF)• มี clients หลายประเภท (Web, Mobile, TV, IoT)
• Client ต้องการ aggregate หลาย APIs
• ต้องการ optimize bandwidth สำหรับ mobile
• Teams แยกกัน (Frontend team owned BFF)

Cautionระวัง

Trade-offs

• More services to maintain
• Duplicated logic across BFFs
• If clients need the same data, use a single API Gateway instead• เพิ่ม services ที่ต้อง maintain
• Duplication logic ระหว่าง BFFs
• ถ้า clients ต้องการข้อมูลเหมือนกัน ใช้ single API Gateway แทน

Tip: if you have many clients but they need the same data — use GraphQL instead. GraphQL lets the client pick the fields it wants without building a separate BFFTip: ถ้ามีหลาย clients แต่ต้องการข้อมูลเหมือนกัน — ใช้ GraphQL แทน GraphQL ให้ client เลือกเอง fields ที่ต้องการ โดยไม่ต้องสร้าง BFF แยก
