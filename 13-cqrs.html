Design Pattern #13

CQRS

Command Query Responsibility Segregation — แยก "การเขียน" และ "การอ่าน" ออกจากกันอย่างสมบูรณ์ เพื่อ Performance และ Scalability สูงสุด

CQRS คืออะไร?

CQRS คือ pattern ที่แบ่งระบบออกเป็น 2 ด้านชัดเจน: Command Side (เขียน/แก้ไขข้อมูล) และ Query Side (อ่านข้อมูล) แต่ละด้านมี model, database, และ handler ของตัวเอง ทำให้ optimize แต่ละด้านได้อิสระ

หลักการพื้นฐานมาจาก CQS (Command Query Separation) ของ Bertrand Meyer: "A function should either change state, or return a value — never both."

⚡ Command Side (Write)
แก้ไขสถานะของระบบ
Commands: CreateOrder, CancelOrder, UpdateStock
Validates business rules
Produces Domain Events
Write-optimized database (normalized)
Returns void / error เท่านั้น
Consistency: strong
Throughput: medium
🔍 Query Side (Read)
คืนค่าโดยไม่แก้ไขข้อมูล
Queries: GetOrder, ListOrders, GetSummary
No business rule validation
Consumes Domain Events → updates projections
Read-optimized database (denormalized)
Returns DTOs / View Models
Consistency: eventual
Throughput: very high

// PlaceOrderCommand — intent to create an order (write, no return data)
type PlaceOrderCommand struct {

domain/order_aggregate.go — Aggregate with events
package domain

import (
	"fmt"
	"time"
)

type OrderStatus string

const (
	StatusPending   OrderStatus = "pending"
	StatusCancelled OrderStatus = "cancelled"
)

// Domain Events — things that happened (past tense)
type OrderPlacedEvent struct {
	OrderID    string
	CustomerID string
	Total      float64
	OccurredAt time.Time
}

type OrderCancelledEvent struct {
	OrderID     string
	Reason      string
	OccurredAt  time.Time
}

// Order aggregate — maintains invariants
type Order struct {
	ID         string
	CustomerID string
	Status     OrderStatus
	Total      float64
	Events     []interface{} // uncommitted events
}

func NewOrder(id, customerID string, total float64) *Order {
	o := &Order{ID: id, CustomerID: customerID, Status: StatusPending, Total: total}
	o.Events = append(o.Events, OrderPlacedEvent{
		OrderID: id, CustomerID: customerID, Total: total,
		OccurredAt: time.Now(),
	})
	return o
}

func (o *Order) Cancel(reason string) error {
	if o.Status == StatusCancelled {
		return fmt.Errorf("order already cancelled")
	}
	o.Status = StatusCancelled
	o.Events = append(o.Events, OrderCancelledEvent{
		OrderID: o.ID, Reason: reason, OccurredAt: time.Now(),
	})
	return nil
}
handler/command_handler.go — Command Side Handler
package handler

import (
	"context"
	"fmt"

	"myapp/command"
	"myapp/domain"
	"myapp/event"
)

type OrderCommandHandler struct {
	writeRepo domain.OrderWriteRepository
	eventBus  event.Bus
}

func NewOrderCommandHandler(repo domain.OrderWriteRepository, bus event.Bus) *OrderCommandHandler {
	return &OrderCommandHandler{writeRepo: repo, eventBus: bus}
}

// HandlePlaceOrder — Command handler returns ONLY error, not data
func (h *OrderCommandHandler) HandlePlaceOrder(ctx context.Context, cmd command.PlaceOrderCommand) error {
	// 1. Calculate total
	var total float64
	for _, item := range cmd.Items {
		total += float64(item.Qty) * item.Price
	}

	// 2. Create aggregate (enforces business rules)
	order := domain.NewOrder(cmd.OrderID, cmd.CustomerID, total)

	// 3. Persist to write database
	if err := h.writeRepo.Save(ctx, order); err != nil {
		return fmt.Errorf("save order: %w", err)
	}

	// 4. Publish domain events → Query side will update projections
	for _, evt := range order.Events {
		if err := h.eventBus.Publish(ctx, evt); err != nil {
			return fmt.Errorf("publish event: %w", err)
		}
	}
	return nil
}

func (h *OrderCommandHandler) HandleCancelOrder(ctx context.Context, cmd command.CancelOrderCommand) error {
	order, err := h.writeRepo.FindByID(ctx, cmd.OrderID)
	if err != nil {
		return err
	}
	if err := order.Cancel(cmd.Reason); err != nil {
		return err
	}
	if err := h.writeRepo.Save(ctx, order); err != nil {
		return err
	}
	for _, evt := range order.Events {
		h.eventBus.Publish(ctx, evt)
	}
	return nil
}
handler/query_handler.go — Query Side Handler
package handler

import (
	"context"

	"myapp/query"
)

// OrderReadRepository — interface for read-optimized database
type OrderReadRepository interface {
	FindSummaryByID(ctx context.Context, id string) (*query.OrderSummaryDTO, error)
	ListByCustomer(ctx context.Context, customerID string, page, pageSize int) ([]*query.OrderSummaryDTO, error)
}

type OrderQueryHandler struct {
	readRepo OrderReadRepository
}

func NewOrderQueryHandler(repo OrderReadRepository) *OrderQueryHandler {
	return &OrderQueryHandler{readRepo: repo}
}

// HandleGetOrder — Query handler returns data (DTO), never modifies state
func (h *OrderQueryHandler) HandleGetOrder(ctx context.Context, q query.GetOrderByIDQuery) (*query.OrderSummaryDTO, error) {
	return h.readRepo.FindSummaryByID(ctx, q.OrderID)
}

func (h *OrderQueryHandler) HandleListOrders(ctx context.Context, q query.ListOrdersByCustomerQuery) ([]*query.OrderSummaryDTO, error) {
	return h.readRepo.ListByCustomer(ctx, q.CustomerID, q.Page, q.PageSize)
}
projector/order_projector.go — Updates Read Model from Events
package projector

import (
	"context"
	"database/sql"
	"log"

	"myapp/domain"
)

// OrderProjector listens to domain events and updates the read model
type OrderProjector struct {
	readDB *sql.DB
}

func NewOrderProjector(readDB *sql.DB) *OrderProjector {
	return &OrderProjector{readDB: readDB}
}

// OnOrderPlaced — updates the denormalized read table
func (p *OrderProjector) OnOrderPlaced(ctx context.Context, evt domain.OrderPlacedEvent) {
	_, err := p.readDB.ExecContext(ctx,
		`INSERT INTO order_summaries (order_id, customer_id, status, total, item_count, created_at)
		 VALUES ($1, $2, 'pending', $3, 0, $4)`,
		evt.OrderID, evt.CustomerID, evt.Total, evt.OccurredAt)
	if err != nil {
		log.Printf("projector OnOrderPlaced error: %v", err)
	}
}

func (p *OrderProjector) OnOrderCancelled(ctx context.Context, evt domain.OrderCancelledEvent) {
	_, err := p.readDB.ExecContext(ctx,
		`UPDATE order_summaries SET status = 'cancelled' WHERE order_id = $1`,
		evt.OrderID)
	if err != nil {
		log.Printf("projector OnOrderCancelled error: %v", err)
	}
}
cmd/main.go — HTTP API wiring Command & Query handlers
package main

import (
	"encoding/json"
	"net/http"
	"time"

	"myapp/command"
	"myapp/handler"
	"myapp/query"
)

func main() {
	// ... setup DBs, repos, event bus ...

	cmdHandler := handler.NewOrderCommandHandler(writeRepo, eventBus)
	qryHandler := handler.NewOrderQueryHandler(readRepo)

	// Command endpoint — returns 202 Accepted, no body
	http.HandleFunc("POST /orders", func(w http.ResponseWriter, r *http.Request) {
		var cmd command.PlaceOrderCommand
		if err := json.NewDecoder(r.Body).Decode(&cmd); err != nil {
			http.Error(w, err.Error(), 400)
			return
		}
		cmd.PlacedAt = time.Now()
		if err := cmdHandler.HandlePlaceOrder(r.Context(), cmd); err != nil {
			http.Error(w, err.Error(), 500)
			return
		}
		w.WriteHeader(http.StatusAccepted) // 202: accepted for processing
	})

	// Query endpoint — returns data (from read model)
	http.HandleFunc("GET /orders/{id}", func(w http.ResponseWriter, r *http.Request) {
		orderID := r.PathValue("id")
		result, err := qryHandler.HandleGetOrder(r.Context(), query.GetOrderByIDQuery{OrderID: orderID})
		if err != nil {
			http.Error(w, err.Error(), 404)
			return
		}
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(result)
	})

	http.ListenAndServe(":8080", nil)
}

เมื่อไหร่ควรใช้ CQRS?

✅
เหมาะกับ
  • Read/Write ratio ไม่สมดุล (read มากกว่า write 10x+)
  • Complex business logic ฝั่ง write, complex reporting ฝั่ง read
  • ต้องการ audit trail / event history สมบูรณ์
  • Scalability — scale read/write tier อิสระ
  • ระบบที่ต้องการ multiple views ของ data เดียวกัน
⚠️
ระวัง
  • Complexity สูงมาก — ไม่เหมาะกับ CRUD อย่างเดียว
  • Eventual consistency — user อาจเห็น data เก่า momentarily
  • ต้องเขียน code เยอะขึ้น — 2 models, 2 databases, projectors
  • Debug ยากกว่า — trace event flow ข้าม services
  • Overkill สำหรับ small/simple applications
