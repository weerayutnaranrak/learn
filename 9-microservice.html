🏗️ Distributed Architecture

Microservice Architecture

Split a large application into small, independent services — deploy separately, scale separately, fail separately, letting large teams work in parallel without conflicts.แบ่ง Application ใหญ่เป็น Services เล็กๆ อิสระ — Deploy แยก, Scale แยก, Fail แยก ทำให้ทีมใหญ่ทำงานขนานกันได้โดยไม่ชนกัน

package main

import (
    "encoding/json"
    "log"
    "net/http"
    "os"
    "time"
)

// ── User Microservice — standalone process ─────────────────────────────────
// มี DB ของตัวเอง, Deploy แยก, Scale แยก

type User struct {
    ID        string    `json:"id"`
    Name      string    `json:"name"`
    Email     string    `json:"email"`
    CreatedAt time.Time `json:"created_at"`
}

// Health check — สำหรับ Load Balancer & Service Discovery
func healthHandler(w http.ResponseWriter, r *http.Request) {
    json.NewEncoder(w).Encode(map[string]string{
        "service": "user-service",
        "status":  "healthy",
        "version": "1.0.0",
    })
}

// GetUser — synchronous API call
func getUserHandler(w http.ResponseWriter, r *http.Request) {
    // ดึง user ID จาก path หรือ header
    userID := r.Header.Get("X-User-ID")
    if userID == "" {
        userID = r.URL.Query().Get("id")
    }

    // ดึงจาก DB ของ User Service
    user := &User{
        ID:        userID,
        Name:      "สมชาย ใจดี",
        Email:     "somchai@example.com",
        CreatedAt: time.Now(),
    }
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(user)
}

func main() {
    port := os.Getenv("PORT")
    if port == "" { port = "3001" }

    // Register กับ Service Discovery (Consul)
    // consul.Register("user-service", port)

    mux := http.NewServeMux()
    mux.HandleFunc("/health", healthHandler)
    mux.HandleFunc("/users",  getUserHandler)

    log.Printf("User Service running on :%s", port)
    log.Fatal(http.ListenAndServe(":"+port, mux))
}

api-gateway/main.go — API GatewayGo
package main

import (
    "context"
    "encoding/json"
    "fmt"
    "log"
    "net/http"
    "net/http/httputil"
    "net/url"
    "os"
    "strings"
    "time"
)

// ── API Gateway — Single Entry Point ──────────────────────────────────────
// ทุก request เข้ามาที่นี่ก่อน ก่อนจะ route ไปหา service ที่เหมาะสม

type ServiceRegistry map[string]string

var services = ServiceRegistry{
    "users":    "http://user-service:3001",
    "orders":   "http://order-service:3002",
    "payments": "http://payment-service:3003",
    "products": "http://product-service:3005",
}

// ── Middleware: Authentication ─────────────────────────────────────────────
func authMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        // Skip auth for health checks
        if r.URL.Path == "/health" {
            next.ServeHTTP(w, r)
            return
        }

        token := r.Header.Get("Authorization")
        if !strings.HasPrefix(token, "Bearer ") {
            http.Error(w, `{"error":"unauthorized"}`, http.StatusUnauthorized)
            return
        }

        // Validate token (จริงๆ ต้อง verify JWT หรือ call auth service)
        userID := validateToken(token[7:])
        if userID == "" {
            http.Error(w, `{"error":"invalid token"}`, http.StatusUnauthorized)
            return
        }

        // Forward user identity ไปยัง downstream services
        r.Header.Set("X-User-ID", userID)
        next.ServeHTTP(w, r)
    })
}

// ── Middleware: Rate Limiting (simple token bucket) ────────────────────────
func rateLimitMiddleware(next http.Handler) http.Handler {
    // Production: ใช้ Redis-based rate limiter
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        // Check rate limit...
        next.ServeHTTP(w, r)
    })
}

// ── Router: ส่ง request ไปยัง microservice ที่ถูกต้อง ─────────────────────
func routeHandler(w http.ResponseWriter, r *http.Request) {
    // /api/users/123 → user-service:3001/users/123
    parts := strings.SplitN(r.URL.Path[5:], "/", 2) // strip /api/
    if len(parts) == 0 || parts[0] == "" {
        http.Error(w, `{"error":"not found"}`, http.StatusNotFound)
        return
    }

    serviceName := parts[0]
    targetBase, ok := services[serviceName]
    if !ok {
        http.Error(w, fmt.Sprintf(`{"error":"service %q not found"}`, serviceName), http.StatusNotFound)
        return
    }

    // Reverse proxy ไปยัง microservice
    target, _ := url.Parse(targetBase)
    proxy := httputil.NewSingleHostReverseProxy(target)
    proxy.ModifyResponse = func(resp *http.Response) error {
        // Add CORS headers
        resp.Header.Set("X-Gateway", "go-api-gateway/1.0")
        return nil
    }
    proxy.ErrorHandler = func(w http.ResponseWriter, r *http.Request, err error) {
        log.Printf("Gateway error for %s: %v", serviceName, err)
        http.Error(w, `{"error":"service unavailable"}`, http.StatusServiceUnavailable)
    }

    // Rewrite path: /api/users → /users
    r.URL.Path = "/" + strings.Join(parts[1:], "/")
    proxy.ServeHTTP(w, r)
}

func validateToken(token string) string {
    // JWT validation ในชีวิตจริง
    if token == "test-token" { return "user-123" }
    return ""
}

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
        json.NewEncoder(w).Encode(map[string]string{"status": "ok"})
    })
    mux.HandleFunc("/api/", routeHandler)

    // Chain middleware
    handler := authMiddleware(rateLimitMiddleware(mux))

    port := os.Getenv("PORT")
    if port == "" { port = "8080" }
    log.Printf("API Gateway running on :%s", port)
    log.Fatal(http.ListenAndServe(":"+port, handler))
}
// api-gateway/index.ts — API Gateway (Express)
import express, { Request, Response, NextFunction } from 'express';
import { createProxyMiddleware } from 'http-proxy-middleware';

// ── API Gateway — Single Entry Point ──────────────────────────────────────
// ทุก request เข้ามาที่นี่ก่อน ก่อนจะ route ไปหา service ที่เหมาะสม

type ServiceRegistry = Record;

const services: ServiceRegistry = {
  users:    'http://user-service:3001',
  orders:   'http://order-service:3002',
  payments: 'http://payment-service:3003',
  products: 'http://product-service:3005',
};

const app = express();

// ── Middleware: Authentication ─────────────────────────────────────────────
function authMiddleware(req: Request, res: Response, next: NextFunction): void {
  if (req.path === '/health') return next();

  const authHeader = req.headers['authorization'] ?? '';
  if (!authHeader.startsWith('Bearer ')) {
    res.status(401).json({ error: 'unauthorized' });
    return;
  }

  // Validate token — in production, verify JWT or call auth service
  const token = authHeader.slice(7);
  const userID = validateToken(token);
  if (!userID) {
    res.status(401).json({ error: 'invalid token' });
    return;
  }

  // Forward user identity to downstream services
  req.headers['x-user-id'] = userID;
  next();
}

function validateToken(token: string): string | null {
  // JWT validation in real life
  return token === 'test-token' ? 'user-123' : null;
}

// ── Middleware: Rate Limiting (simple counter) ─────────────────────────────
// Production: use Redis-based rate limiter (e.g. express-rate-limit + ioredis)
function rateLimitMiddleware(_req: Request, _res: Response, next: NextFunction): void {
  // Check rate limit...
  next();
}

app.use(authMiddleware);
app.use(rateLimitMiddleware);

app.get('/health', (_req, res) => res.json({ status: 'ok' }));

// ── Router: proxy /api//... → upstream microservice ───────────────
app.use('/api/:serviceName', (req: Request, res: Response, next: NextFunction) => {
  const { serviceName } = req.params;
  const target = services[serviceName];
  if (!target) {
    res.status(404).json({ error: `service "${serviceName}" not found` });
    return;
  }

  // Strip /api/ prefix and proxy to upstream
  createProxyMiddleware({
    target,
    changeOrigin: true,
    pathRewrite: { [`^/api/${serviceName}`]: '' },
    on: {
      proxyRes: (proxyRes) => {
        proxyRes.headers['x-gateway'] = 'ts-api-gateway/1.0';
      },
      error: (_err, _req, res) => {
        (res as Response).status(503).json({ error: 'service unavailable' });
      },
    },
  })(req, res, next);
});

const PORT = process.env.PORT ?? '8080';
app.listen(Number(PORT), () => console.log(`API Gateway running on :${PORT}`));
order-service/client.go — Inter-service HTTP Client with Circuit BreakerGo
package main

import (
    "context"
    "encoding/json"
    "fmt"
    "net/http"
    "sync/atomic"
    "time"
)

// ── Circuit Breaker ────────────────────────────────────────────────────────
// ป้องกัน cascade failure: ถ้า User Service ล้ม → Order Service ไม่ล้มตาม

type CircuitState int32
const (
    StateClosed   CircuitState = 0 // ปกติ — ส่ง request ได้
    StateOpen     CircuitState = 1 // เปิด — ไม่ส่ง request (fail fast)
    StateHalfOpen CircuitState = 2 // ทดสอบว่าปกติแล้วหรือยัง
)

type CircuitBreaker struct {
    state       atomic.Int32
    failures    atomic.Int32
    maxFailures int32
    resetAfter  time.Duration
    lastFailure time.Time
}

func NewCircuitBreaker(maxFailures int32, resetAfter time.Duration) *CircuitBreaker {
    return &CircuitBreaker{maxFailures: maxFailures, resetAfter: resetAfter}
}

func (cb *CircuitBreaker) CanRequest() bool {
    state := CircuitState(cb.state.Load())
    switch state {
    case StateClosed:
        return true
    case StateOpen:
        if time.Since(cb.lastFailure) > cb.resetAfter {
            cb.state.Store(int32(StateHalfOpen))
            return true
        }
        return false
    case StateHalfOpen:
        return true
    }
    return true
}

func (cb *CircuitBreaker) RecordSuccess() {
    cb.failures.Store(0)
    cb.state.Store(int32(StateClosed))
}

func (cb *CircuitBreaker) RecordFailure() {
    fails := cb.failures.Add(1)
    cb.lastFailure = time.Now()
    if fails >= cb.maxFailures {
        cb.state.Store(int32(StateOpen))
    }
}

// ── UserServiceClient — ใช้ใน Order Service ──────────────────────────────
type UserServiceClient struct {
    baseURL string
    http    *http.Client
    cb      *CircuitBreaker
}

func NewUserServiceClient(baseURL string) *UserServiceClient {
    return &UserServiceClient{
        baseURL: baseURL,
        http:    &http.Client{Timeout: 3 * time.Second},
        cb:      NewCircuitBreaker(5, 30*time.Second),
    }
}

type UserInfo struct {
    ID    string `json:"id"`
    Name  string `json:"name"`
    Email string `json:"email"`
}

func (c *UserServiceClient) GetUser(ctx context.Context, userID string) (*UserInfo, error) {
    // Circuit breaker check
    if !c.cb.CanRequest() {
        return nil, fmt.Errorf("circuit open: user-service unavailable")
    }

    req, err := http.NewRequestWithContext(ctx, "GET",
        fmt.Sprintf("%s/users?id=%s", c.baseURL, userID), nil)
    if err != nil {
        return nil, err
    }

    resp, err := c.http.Do(req)
    if err != nil {
        c.cb.RecordFailure()
        return nil, fmt.Errorf("user-service request failed: %w", err)
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusOK {
        c.cb.RecordFailure()
        return nil, fmt.Errorf("user-service returned %d", resp.StatusCode)
    }

    var user UserInfo
    if err := json.NewDecoder(resp.Body).Decode(&user); err != nil {
        return nil, err
    }

    c.cb.RecordSuccess()
    return &user, nil
}
✅ Summary — When Should You Use Microservices?สรุป — เมื่อไหร่ควรใช้ Microservices?
✅
Use Whenควรใช้เมื่อ
  • Large team (10+ people) needs autonomyทีมใหญ่ (10+ คน) ต้องการ autonomy
  • Services need to scale differently (checkout ≠ browse)Services ต้อง scale ต่างกัน (checkout ≠ browse)
  • Different technology stacks per areaTechnology stack แต่ละส่วนต่างกัน
  • System requires high availability per componentSystem ต้องการ high availability แต่ละส่วน
⚠️
Avoid Whenไม่ควรใช้เมื่อ
  • Small startup (1–5 person team)Startup ขนาดเล็ก (ทีม 1-5 คน)
  • Domain boundaries aren't clear yetยังไม่รู้ domain boundaries ชัดเจน
  • No infrastructure ready (K8s, monitoring)ไม่มี infrastructure ที่พร้อม (K8s, monitoring)
  • "Premature optimization" — start with a Monolith first"Premature optimization" — เริ่ม Monolith ก่อน
Martin Fowler's Advice: "Don't start with microservices. Start with a monolith, understand your domain, then extract services when you need to." — Don't over-engineer from the start.อย่า over-engineer ตั้งแต่ต้น
