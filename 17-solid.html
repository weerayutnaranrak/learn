SOLID Principles

5 หลักการออกแบบ Object-Oriented ที่ทำให้ code ยืดหยุ่น ขยายได้ง่าย และทดสอบได้ — รากฐานของ Clean Architecture, Hexagonal, และ Pattern ทุกอย่างที่เราเรียนมา

S

เพิ่ม feature ใหม่ = เพิ่ม code ใหม่ ไม่แก้ code เดิม
L
Liskov Substitution
เปลี่ยน subclass แทน parent class ได้เลย โดย program ยังทำงานถูกต้อง
I
Interface Segregation
แยก interface ให้เล็กตรงประเด็น — ไม่บังคับ implement method ที่ตัวเองไม่ต้องการ
D
Dependency Inversion
ทุก module ขึ้นต่อกันผ่าน interface ไม่ใช่เชื่อมตรงกับ implementation จริงๆ
ทำไม SOLID ถึงสำคัญ? หลักการเหล่านี้เป็น รากฐาน ของทุก Architecture Pattern ที่เรียนมา — Clean Architecture ใช้ DIP, Hexagonal ใช้ DIP + ISP, Repository Pattern ใช้ DIP, Strategy Pattern ใช้ OCP ทั้งสิ้น
S
Single Responsibility Principle
A class should have only one reason to change
หลักการ: 1 module ควรมีเหตุผลเดียวในการเปลี่ยน — หมายความว่า มี stakeholder เดียว ที่เป็นเจ้าของ responsibility นั้น
สัญญาณว่าผิด SRP: เมื่อมีเหตุผล ≥ 2 อย่างที่ทำให้ต้องแก้ไข class เดิม (เช่น เปลี่ยน business rule และ เปลี่ยน email template)
❌ ละเมิด SRP — หนึ่ง service ทำทุกอย่าง ผิดหลักการ 
// OrderService ทำ 3 อย่างพร้อมกัน:
// 1) save order  2) send email  3) gen PDF
// ถ้าอยากเปลี่ยน email template หรือ PDF layout
// ต้องมาแก้ไฟล์นี้ทั้งที่ไม่เกี่ยวกับ order logic

type OrderService struct {
    db *sql.DB
}

func (s *OrderService) PlaceOrder(o Order) error {
    // business logic
    _, err := s.db.Exec(
        `INSERT INTO orders VALUES(?,?)`, o.ID, o.Total)
    if err != nil {
        return err
    }

    // ส่ง email — ไม่ใช่หน้าที่ OrderService!
    auth := smtp.PlainAuth("", "from@co.th", "pass", "smtp.co.th")
    smtp.SendMail("smtp.co.th:587", auth,
        "from@co.th", []string{o.Email},
        []byte("Order confirmed: "+o.ID))

    // สร้าง PDF invoice — ไม่ใช่หน้าที่ OrderService!
    pdf := gofpdf.New("P", "mm", "A4", "")
    pdf.AddPage()
    pdf.SetFont("Arial", "B", 16)
    pdf.Cell(40, 10, "Invoice #"+o.ID)
    pdf.OutputFileAndClose("invoice.pdf")

    return nil
}
✅ ถูกต้อง — แยก responsibility ถูกหลักการ 
// แต่ละ type มีเหตุผลเดียวในการเปลี่ยน

type OrderRepository struct{ db *sql.DB }

func (r *OrderRepository) Save(o Order) error {
    _, err := r.db.Exec(
        `INSERT INTO orders VALUES(?,?)`, o.ID, o.Total)
    return err
}

// เหตุผลเดียวที่จะเปลี่ยน EmailService:
// เปลี่ยน email template หรือ SMTP provider
type EmailService struct{ smtpAddr string }

func (e *EmailService) SendConfirmation(o Order) error {
    body := "Order confirmed: " + o.ID
    return smtp.SendMail(e.smtpAddr, nil,
        "no-reply@co.th", []string{o.Email},
        []byte(body))
}

// เปลี่ยน layout หรือ format ของ PDF
type InvoiceService struct{}

func (i *InvoiceService) Generate(o Order) ([]byte, error) {
    // ... PDF generation
    return nil, nil

// OrderService: orchestrate เท่านั้น
type OrderService struct {
    repo    *OrderRepository
    email   *EmailService
    invoice *InvoiceService
}

func (s *OrderService) PlaceOrder(o Order) error {
    if err := s.repo.Save(o); err != nil {
        return err
    }
    go s.email.SendConfirmation(o) // async
    return nil
}
เคล็ดลับ: ถามตัวเองว่า "ถ้าต้องเปลี่ยน X จะต้องแก้ไฟล์นี้ไหม?" ถ้าตอบ "ใช่" กับเหตุผลหลายอย่าง — แยก type ออกมา
O
Open / Closed Principle
Open for extension, closed for modification
หลักการ: เพิ่ม feature ใหม่ได้โดย เพิ่ม code ใหม่ ไม่ใช่แก้ code เดิม — ลด risk ของ regression bug
เครื่องมือ: Interface + Polymorphism — ให้ caller depend on abstraction แล้ว inject implementation ใหม่ได้เสมอ
❌ ละเมิด OCP — เพิ่ม type ต้องแก้ switch ผิดหลักการ 
type DiscountService struct{}

// เพิ่ม user type ใหม่ทุกครั้ง
// ต้องมาแก้ function นี้ → ผิด OCP
func (d *DiscountService) Calculate(
    total float64, userType string,
) float64 {
    switch userType {
    case "vip":
        return total * 0.20
    case "member":
        return total * 0.10
    case "new_user":
        return total * 0.05
    // เพิ่ม "partner" → ต้องแก้ที่นี่
    // เพิ่ม "staff"   → ต้องแก้ที่นี่
    // แก้ทุกครั้งมีโอกาส break case เดิม
    default:
        return 0
    }
}
✅ ถูกต้อง — เพิ่ม type ใหม่โดยไม่แก้ code เดิม ถูกหลักการ 
// Abstraction ที่ "ปิด" ต่อการแก้ไข
type DiscountStrategy interface {
    Discount(total float64) float64
}

// ขยายได้โดยเพิ่ม type ใหม่
type VIPDiscount     struct{}
type MemberDiscount  struct{}
type NewUserDiscount struct{}
type PartnerDiscount struct{ Rate float64 } // ใหม่

func (VIPDiscount)     Discount(t float64) float64 { return t * 0.20 }
func (MemberDiscount)  Discount(t float64) float64 { return t * 0.10 }
func (NewUserDiscount) Discount(t float64) float64 { return t * 0.05 }
func (p PartnerDiscount) Discount(t float64) float64 { return t * p.Rate }

// OrderService ไม่ต้องเปลี่ยนเลย
// เพิ่ม discount ใหม่ → เพิ่ม struct ใหม่อย่างเดียว
type OrderService struct {
    discount DiscountStrategy
}

func (s *OrderService) FinalPrice(total float64) float64 {
    return total - s.discount.Discount(total)
}

// inject ตอน wire-up
svc := &OrderService{discount: VIPDiscount{}}
svc2 := &OrderService{discount: PartnerDiscount{Rate: 0.15}}
ความเชื่อมโยง: OCP คือเหตุผลที่เราใช้ Strategy Pattern, Plugin Pattern, และ Middleware chains — ทุก pattern เหล่านี้เป็นการ extend behavior โดยไม่แก้ core code
L
Liskov Substitution Principle
Subtypes must be substitutable for their base types without altering correctness
หลักการ: ถ้า code ทำงานกับ Shape ได้ถูกต้อง ก็ต้องทำงานกับ Circle (ซึ่ง implement Shape) ได้ถูกต้องเช่นกัน โดยไม่ต้อง type-assert
สัญญาณว่าผิด LSP: มี if _, ok := x.(ConcreteType) อยู่ในโค้ด — นั่นหมายความว่า interface ออกแบบมาไม่ดีหรือ subtype ไม่ได้ honor contract
❌ ละเมิด LSP — Square ทำ Rectangle ไม่ได้ ผิดหลักการ 
type Resizable interface {
    SetWidth(w float64)
    SetHeight(h float64)
    Area() float64
}

type Rectangle struct{ W, H float64 }
func (r *Rectangle) SetWidth(w float64)  { r.W = w }
func (r *Rectangle) SetHeight(h float64) { r.H = h }
func (r *Rectangle) Area() float64       { return r.W * r.H }

// Square "is-a" Rectangle? ไม่จริง!
type Square struct{ Side float64 }
func (s *Square) SetWidth(w float64)  { s.Side = w } // override both
func (s *Square) SetHeight(h float64) { s.Side = h } // ทำให้ Width เปลี่ยนด้วย!
func (s *Square) Area() float64       { return s.Side * s.Side }

// caller คาดหวัง behavior แบบ Rectangle
func resize(r Resizable) {
    r.SetWidth(5)
    r.SetHeight(3)
    // Rectangle → Area = 15 ✓
    // Square    → Area = 9  ✗ (SetHeight override Side = 3)
    fmt.Println("Area:", r.Area())
}
✅ ถูกต้อง — interface ตรงกับ behavior จริง ถูกหลักการ 
// แยก interface ตาม capability จริงๆ
type Shape interface {
    Area() float64
}

type Resizable interface {
    Shape
    SetWidth(w float64)
    SetHeight(h float64)
}

// Rectangle implement Resizable (มี SetWidth + SetHeight)
type Rectangle struct{ W, H float64 }
func (r *Rectangle) SetWidth(w float64)  { r.W = w }
func (r *Rectangle) SetHeight(h float64) { r.H = h }
func (r *Rectangle) Area() float64       { return r.W * r.H }

// Square implement แค่ Shape — honest!
// ไม่ implement Resizable เพราะ SetWidth ≠ SetHeight ไม่ได้
type Square struct{ Side float64 }
func (s *Square) SetSide(n float64)  { s.Side = n }
func (s *Square) Area() float64      { return s.Side * s.Side }

// function รับแค่ interface ที่จำเป็น
func printArea(s Shape) {
    fmt.Println("Area:", s.Area()) // ทั้ง Rectangle และ Square ใช้ได้
}

func resize(r Resizable, w, h float64) {
    r.SetWidth(w)
    r.SetHeight(h)
    // Square ไม่ผ่านมาที่นี่ → ไม่มี LSP violation
}
Rule of Thumb: ถ้าต้อง type-switch หรือ type-assert บน interface ที่รับมา — เป็นสัญญาณว่า interface ออกแบบมาไม่ตรงกับ real contract ให้ split interface หรือ redesign hierarchy
I
Interface Segregation Principle
Clients should not be forced to depend on interfaces they do not use
หลักการ: Interface ควรเล็กและตรงประเด็น — อย่าสร้าง "fat interface" ที่บังคับให้ทุก implementor ต้อง implement method ที่ตัวเองไม่ต้องการใช้
ใน Go: ISP มาโดยธรรมชาติ — Go interface เป็น implicit และ define ที่ผู้ใช้ ไม่ใช่ผู้สร้าง ทำให้ง่ายต่อการแยก interface เล็กๆ
❌ ละเมิด ISP — Fat interface บังคับ implement ทุก method ผิดหลักการ 
// OrderRepository ใหญ่เกินไป
// บางคนอยากแค่ Read แต่ถูกบังคับ implement Save, Delete ด้วย
type OrderRepository interface {
    FindByID(id string) (*Order, error)
    FindByUserID(uid string) ([]*Order, error)
    Save(o *Order) error
    Delete(id string) error
    GenerateReport() ([]byte, error) // ไม่ใช่หน้าที่ repository!
    ExportCSV() (string, error)      // ไม่ใช่หน้าที่ repository!
}

// ReadOnlyService ต้องการแค่ FindByID
// แต่ถูกบังคับ implement Save, Delete, Report ด้วย!
type CacheRepo struct{}

func (c *CacheRepo) FindByID(id string) (*Order, error)     { /* cache lookup */ }
func (c *CacheRepo) FindByUserID(uid string) ([]*Order, error) { /* ... */ }
func (c *CacheRepo) Save(o *Order) error                    { return errors.New("read-only") }
func (c *CacheRepo) Delete(id string) error                 { return errors.New("read-only") }
func (c *CacheRepo) GenerateReport() ([]byte, error)        { return nil, errors.New("N/A") }
func (c *CacheRepo) ExportCSV() (string, error)             { return "", errors.New("N/A") }
✅ ถูกต้อง — แยก interface ตาม role ถูกหลักการ 
// Interface เล็กๆ แยกตาม role
type OrderReader interface {
    FindByID(id string) (*Order, error)
    FindByUserID(uid string) ([]*Order, error)
}

type OrderWriter interface {
    Save(o *Order) error
    Delete(id string) error
}

// Compose เมื่อต้องการทั้งสองอย่าง
type OrderRepository interface {
    OrderReader
    OrderWriter
}

// CacheRepo implement แค่ที่ตัวเองทำได้จริง
type CacheRepo struct{ data map[string]*Order }

func (c *CacheRepo) FindByID(id string) (*Order, error)        { return c.data[id], nil }
func (c *CacheRepo) FindByUserID(uid string) ([]*Order, error) { /* ... */ }
// ไม่ต้อง implement Save/Delete เพราะไม่ได้ implement OrderWriter

// Service ขอแค่ที่ตัวเองต้องการ
type ReportService struct {
    reader OrderReader // ต้องการแค่ Read — ไม่ต้องรู้ Save/Delete
}

type OrderService struct {
    repo OrderRepository // ต้องการทั้ง Read และ Write
}
Go tip: กำหนด interface ที่ผู้ใช้ ไม่ใช่ผู้สร้าง — io.Reader (1 method), io.Writer (1 method), io.ReadWriter (compose) คือตัวอย่างที่ดีที่สุดของ ISP ใน Go standard library
D
Dependency Inversion Principle
High-level modules should not depend on low-level modules — both should depend on abstractions
หลักการ: High-level module (business logic) ไม่ควร import low-level module (DB, HTTP) โดยตรง — ทั้งคู่ควร depend on interface
ผล: สามารถ swap implementation ได้โดยไม่กระทบ business logic — ทดสอบ business logic ด้วย mock ได้ทันที
❌ ละเมิด DIP — High-level depend on Low-level ผิดหลักการ 
// Low-level module: concrete DB implementation
type PostgresOrderRepo struct{ db *sql.DB }

func (r *PostgresOrderRepo) Save(o Order) error {
    _, err := r.db.Exec(`INSERT INTO orders...`)
    return err
}

// High-level module: business logic
// ผิดตรงที่ depend on concrete type โดยตรง
type OrderService struct {
    repo *PostgresOrderRepo // ← concrete! ผิด DIP
}

// สร้าง dependency เอง — ยิ่งผิดหนัก
func NewOrderService() *OrderService {
    db, _ := sql.Open("postgres", os.Getenv("DB_URL"))
    return &OrderService{
        repo: &PostgresOrderRepo{db: db}, // tightly coupled
    }
}

✅ ถูกต้อง — ทั้งสอง depend on abstraction ถูกหลักการ 
// Abstraction — กำหนดโดย High-level module
type OrderStorer interface {
    Save(o Order) error
    FindByID(id string) (*Order, error)
}

// High-level module: depend on interface เท่านั้น
type OrderService struct {
    store OrderStorer // ← abstraction!
}

// รับ dependency จาก outside (Dependency Injection)
func NewOrderService(store OrderStorer) *OrderService {
    return &OrderService{store: store}
}

func (s *OrderService) PlaceOrder(o Order) error {
    return s.store.Save(o)
}

// Low-level modules: implement interface
type PostgresOrderRepo struct{ db *sql.DB }
func (r *PostgresOrderRepo) Save(o Order) error {/* ... */}
func (r *PostgresOrderRepo) FindByID(id string) (*Order, error) {/* ... */}

// Mock สำหรับ test
type MockOrderRepo struct{ saved []Order }
func (m *MockOrderRepo) Save(o Order) error { m.saved = append(m.saved, o); return nil }
func (m *MockOrderRepo) FindByID(id string) (*Order, error) { return &Order{ID: id}, nil }

// --- main.go ---
repo := &PostgresOrderRepo{db: db}
svc  := NewOrderService(repo)

// --- test ---
mock := &MockOrderRepo{}
svc  := NewOrderService(mock) // ไม่ต้องการ DB เลย!
DIP = รากฐานของ Clean Architecture: Dependency Rule ที่ว่า "source code dependencies must point inward" คือ DIP นั่นเอง — Use Case depend on Repository interface, Infrastructure implement interface นั้น

// ── D: Abstraction ที่ High-level กำหนด ──────────────────────
type OrderRepository interface {      // ISP: เล็ก ตรงประเด็น
    Save(o *Order) error
    FindByID(id string) (*Order, error)
}

type Notifier interface {             // ISP: แยกจาก Repository
    Notify(email, msg string) error
}

// ── S: OrderService มีหน้าที่เดียว — orchestrate ──────────────
type OrderService struct {
    repo     OrderRepository  // D: depend on abstraction
    notifier Notifier         // D: depend on abstraction
}

func NewOrderService(r OrderRepository, n Notifier) *OrderService {
    return &OrderService{repo: r, notifier: n}
}

func (s *OrderService) PlaceOrder(o *Order) error {
    if err := s.repo.Save(o); err != nil {  // ไม่รู้ว่าเป็น Postgres หรือ Mock
        return err
    }
    go s.notifier.Notify(o.UserEmail, "Order placed!")
    return nil
}

// ── O: เพิ่ม discount strategy โดยไม่แก้ OrderService ──────────
type DiscountStrategy interface {
    Discount(total float64) float64
}

type VIPDiscount    struct{}
type StaffDiscount  struct{ Rate float64 } // ใหม่ — ไม่ต้องแก้ที่ไหนเลย

func (VIPDiscount) Discount(t float64) float64  { return t * 0.20 }
func (s StaffDiscount) Discount(t float64) float64 { return t * s.Rate }

// ── L: PaymentProcessor ต้อง honor contract ────────────────────
type PaymentProcessor interface {
    Pay(amount float64) error
}

// CreditCard และ PromptPay ต่าง implement Pay() ได้ถูกต้อง
// caller ไม่ต้องรู้ว่าเป็น type ไหน — ไม่มี type-switch
type CreditCardProcessor struct{ cardNo string }
type PromptPayProcessor   struct{ qrCode string }

func (c *CreditCardProcessor) Pay(amount float64) error { /* charge card */ return nil }
func (p *PromptPayProcessor)  Pay(amount float64) error { /* scan QR */    return nil }

func checkout(p PaymentProcessor, amount float64) error {
    return p.Pay(amount) // LSP: ทั้ง CreditCard และ PromptPay ใช้แทนกันได้
}
🏗️ SOLID กับ Architecture Patterns
แต่ละ pattern ที่เรียนมาใช้ SOLID อย่างไร
🏛️ Clean / Onion Architecture
D — Use Case depend on Repository interface
S — Entity, Use Case, Adapter แยกกัน
I — Repository interface เล็กและตรงประเด็น
⬡ Hexagonal Architecture
D — Core depend on Ports (interfaces)
I — Primary Port ≠ Secondary Port
O — เพิ่ม Adapter ใหม่ ไม่แก้ Core
🗄️ Repository Pattern
D — Service depend on Repository interface
S — Repository มีหน้าที่เดียวคือ data access
L — MockRepo แทน PostgresRepo ได้ใน test
🔄 CQRS
S — Command Handler ≠ Query Handler
I — CommandBus ≠ QueryBus (interface แยก)
O — เพิ่ม Event Handler ใหม่ ไม่แก้เดิม
⚡ Event-Driven
O — เพิ่ม Event Handler ใหม่ ไม่แก้ Publisher
S — Publisher มีหน้าที่เดียว ไม่รู้จัก Consumer
D — Depend on EventBus interface
🏗️ Microservices
S — แต่ละ service มี 1 bounded context
O — เพิ่ม service ใหม่ ไม่แก้ service เดิม
D — สื่อสารผ่าน API contract (interface)
