Layered / N-Tier Architecture

แบ่ง Application ออกเป็นชั้น (Layers) ที่มีหน้าที่ชัดเจน แต่ละชั้นสื่อสารกับชั้นถัดไปเท่านั้น — สถาปัตยกรรมที่นิยมที่สุดสำหรับ Business Applications

💡 Layered Architecture คืออะไร?
Organization of code into distinct horizontal layers with clear responsibilities

แนวคิดหลักคือ Separation of Concerns — แต่ละ Layer มีหน้าที่ของตัวเอง และ depend เฉพาะ Layer ที่อยู่ด้านล่าง (Dependency Rule: บนลงล่าง)

domain/post.go — Business Layer (Domain)Go
package domain

import (
    "errors"
    "strings"
    "time"
)

// ── Business Layer: Domain Objects และ Rules ──────────────────────────────
// ไม่รู้จัก HTTP, Database, หรือ Framework ใดๆ

type PostID int64
type UserID int64

type Post struct {
    ID        PostID
    Title     string
    Body      string
    AuthorID  UserID
    Tags      []string
    Published bool
    CreatedAt time.Time
    UpdatedAt time.Time
}

// Business Rule: Title ต้องไม่ว่างและไม่เกิน 200 ตัวอักษร
func (p *Post) SetTitle(title string) error {
    title = strings.TrimSpace(title)
    if title == "" {
        return errors.New("title cannot be empty")
    }
    if len(title) > 200 {
        return errors.New("title must not exceed 200 characters")
    }
    p.Title = title
    return nil
}

// Business Rule: publish ได้เมื่อมี body เท่านั้น
func (p *Post) Publish() error {
    if strings.TrimSpace(p.Body) == "" {
        return errors.New("cannot publish post without body")
    }
    p.Published = true
    p.UpdatedAt = time.Now()
    return nil
}

// Repository Interface — Business Layer กำหนด, Data Layer implement
type PostRepository interface {
    Create(post *Post) (*Post, error)
    FindByID(id PostID) (*Post, error)
    FindPublished(limit, offset int) ([]*Post, error)
    Update(post *Post) (*Post, error)
    Delete(id PostID) error
}

// ── Business Service ────────────────────────────────────────────────────────
type PostService struct {
    repo PostRepository
}

func NewPostService(repo PostRepository) *PostService {
    return &PostService{repo: repo}
}

func (s *PostService) CreateDraft(title, body string, authorID UserID, tags []string) (*Post, error) {
    post := &Post{
        Body:      body,
        AuthorID:  authorID,
        Tags:      tags,
        CreatedAt: time.Now(),
        UpdatedAt: time.Now(),
    }
    if err := post.SetTitle(title); err != nil { // Business rule
        return nil, err
    }
    return s.repo.Create(post)
}

func (s *PostService) PublishPost(id PostID) (*Post, error) {
    post, err := s.repo.FindByID(id)
    if err != nil {
        return nil, err
    }
    if err := post.Publish(); err != nil { // Business rule
        return nil, err
    }
    return s.repo.Update(post)
}

func (s *PostService) GetFeed(page, pageSize int) ([]*Post, error) {
    if page < 1 { page = 1 }
    if pageSize < 1 || pageSize > 50 { pageSize = 20 }
    offset := (page - 1) * pageSize
    return s.repo.FindPublished(pageSize, offset)
}
dataaccess/postgres_repo.go — Data Access LayerGo
package dataaccess

import (
    "database/sql"
    "fmt"
    "myapp/domain"
    "time"
)

// ── Data Access Layer: implements domain.PostRepository ────────────────────
// รู้จัก PostgreSQL, SQL — Business Layer ไม่รู้
type PostgresPostRepository struct {
    db *sql.DB
}

func NewPostgresPostRepository(db *sql.DB) domain.PostRepository {
    return &PostgresPostRepository{db: db}
}

func (r *PostgresPostRepository) Create(post *domain.Post) (*domain.Post, error) {
    row := r.db.QueryRow(`
        INSERT INTO posts (title, body, author_id, published, created_at, updated_at)
        VALUES ($1, $2, $3, $4, $5, $6)
        RETURNING id`,
        post.Title, post.Body, post.AuthorID,
        post.Published, post.CreatedAt, post.UpdatedAt,
    )
    return post, row.Scan(&post.ID)
}

func (r *PostgresPostRepository) FindByID(id domain.PostID) (*domain.Post, error) {
    var p domain.Post
    err := r.db.QueryRow(
        `SELECT id,title,body,author_id,published,created_at,updated_at FROM posts WHERE id=$1`, id,
    ).Scan(&p.ID, &p.Title, &p.Body, &p.AuthorID, &p.Published, &p.CreatedAt, &p.UpdatedAt)
    if err == sql.ErrNoRows {
        return nil, fmt.Errorf("post %d not found", id)
    }
    return &p, err
}

func (r *PostgresPostRepository) FindPublished(limit, offset int) ([]*domain.Post, error) {
    rows, err := r.db.Query(
        `SELECT id,title,body,author_id,created_at FROM posts WHERE published=true ORDER BY created_at DESC LIMIT $1 OFFSET $2`,
        limit, offset,
    )
    if err != nil { return nil, err }
    defer rows.Close()
    var posts []*domain.Post
    for rows.Next() {
        var p domain.Post
        rows.Scan(&p.ID, &p.Title, &p.Body, &p.AuthorID, &p.CreatedAt)
        posts = append(posts, &p)
    }
    return posts, nil
}

func (r *PostgresPostRepository) Update(post *domain.Post) (*domain.Post, error) {
    post.UpdatedAt = time.Now()
    _, err := r.db.Exec(
        `UPDATE posts SET title=$1,body=$2,published=$3,updated_at=$4 WHERE id=$5`,
        post.Title, post.Body, post.Published, post.UpdatedAt, post.ID,
    )
    return post, err
}

func (r *PostgresPostRepository) Delete(id domain.PostID) error {
    _, err := r.db.Exec(`DELETE FROM posts WHERE id=$1`, id)
    return err
}
presentation/handler.go — Presentation Layer (HTTP)Go
package presentation

import (
    "encoding/json"
    "net/http"
    "strconv"
    "myapp/domain"
)

// ── Presentation Layer: HTTP Handlers ──────────────────────────────────────
// รู้จัก HTTP แต่ไม่รู้จัก Database
// เรียก Business Layer ผ่าน Service

type PostHandler struct {
    svc *domain.PostService
}

func NewPostHandler(svc *domain.PostService) *PostHandler {
    return &PostHandler{svc: svc}
}

// Request/Response DTOs — ไม่ใช่ Domain objects โดยตรง
type CreatePostRequest struct {
    Title string   `json:"title"`
    Body  string   `json:"body"`
    Tags  []string `json:"tags"`
}

type PostResponse struct {
    ID        int64    `json:"id"`
    Title     string   `json:"title"`
    Body      string   `json:"body"`
    Tags      []string `json:"tags"`
    Published bool     `json:"published"`
    CreatedAt string   `json:"created_at"`
}

func toPostResponse(p *domain.Post) PostResponse {
    return PostResponse{
        ID: int64(p.ID), Title: p.Title, Body: p.Body,
        Tags: p.Tags, Published: p.Published,
        CreatedAt: p.CreatedAt.Format("2006-01-02T15:04:05Z"),
    }
}

func (h *PostHandler) CreatePost(w http.ResponseWriter, r *http.Request) {
    var req CreatePostRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        writeError(w, http.StatusBadRequest, "invalid request body")
        return
    }

    // เรียก Business Layer (ไม่ใช่ DB โดยตรง!)
    post, err := h.svc.CreateDraft(req.Title, req.Body, 1, req.Tags)
    if err != nil {
        writeError(w, http.StatusBadRequest, err.Error())
        return
    }

    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(http.StatusCreated)
    json.NewEncoder(w).Encode(toPostResponse(post))
}

func (h *PostHandler) PublishPost(w http.ResponseWriter, r *http.Request) {
    idStr := r.URL.Query().Get("id")
    id, err := strconv.ParseInt(idStr, 10, 64)
    if err != nil {
        writeError(w, http.StatusBadRequest, "invalid post id")
        return
    }
    post, err := h.svc.PublishPost(domain.PostID(id))
    if err != nil {
        writeError(w, http.StatusBadRequest, err.Error())
        return
    }
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(toPostResponse(post))
}

func (h *PostHandler) GetFeed(w http.ResponseWriter, r *http.Request) {
    page, _ := strconv.Atoi(r.URL.Query().Get("page"))
    posts, err := h.svc.GetFeed(page, 20)
    if err != nil {
        writeError(w, http.StatusInternalServerError, "failed to get feed")
        return
    }
    result := make([]PostResponse, len(posts))
    for i, p := range posts { result[i] = toPostResponse(p) }
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(result)
}

func writeError(w http.ResponseWriter, code int, msg string) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(code)
    json.NewEncoder(w).Encode(map[string]string{"error": msg})
}
cmd/main.go — Wire all 3 layers togetherGo
package main

import (
    "database/sql"
    "log"
    "net/http"
    "myapp/dataaccess"
    "myapp/domain"
    "myapp/presentation"
    _ "github.com/lib/pq"
)

func main() {
    // Layer 3: Data Access
    db, _ := sql.Open("postgres", "postgres://localhost/blog")

    // Layer 2: Business
    repo := dataaccess.NewPostgresPostRepository(db)
    svc  := domain.NewPostService(repo)

    // Layer 1: Presentation
    handler := presentation.NewPostHandler(svc)

    mux := http.NewServeMux()
    mux.HandleFunc("POST /posts",        handler.CreatePost)
    mux.HandleFunc("POST /posts/publish", handler.PublishPost)
    mux.HandleFunc("GET  /feed",         handler.GetFeed)

    log.Println("Blog API running on :8080")
    log.Fatal(http.ListenAndServe(":8080", mux))
}

✅
เหมาะกับ
  • Business Applications ทั่วไป
  • ทีมที่เพิ่งเริ่มต้น — เข้าใจง่าย
  • Monolithic applications
  • Projects ที่ต้องการ structure ชัดเจน
⚠️
ข้อจำกัด
  • อาจ scale ยากถ้าโต (จำเป็นต้อง scale ทั้งหมด)
  • Strict layer → request ต้องผ่านทุก layer
  • ไม่เหมาะกับ high-performance scenarios
  • Dependency ชี้ลงเท่านั้น — ระวัง cross-layer calls
Anti-pattern ที่พบบ่อย: "Layer Skipping" — Presentation เรียก Data Access โดยตรง (ข้าม Business Layer) ทำให้ Business Rules กระจัดกระจาย ควรบังคับให้ทุก call ผ่าน Business Layer เสมอ
