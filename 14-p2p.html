Architecture Pattern #14

Peer-to-Peer Architecture

Every Node acts as both Client and Server — no Central Authority. Distributed, Resilient, and Scalable.ทุก Node ทำหน้าที่เป็นทั้ง Client และ Server — ไม่มี Central Authority ระบบกระจาย Resilient และ Scalable

What is P2P?P2P คืออะไร?

Peer-to-Peer (P2P) is an architectural pattern where every node in the system (peer) is equal — it can act as a client (requesting resources) and a server (providing resources) simultaneously, without relying on a central server.Peer-to-Peer (P2P) คือ architectural pattern ที่แต่ละ node ในระบบ (peer) มีความสามารถเท่ากัน — สามารถทำหน้าที่เป็น client (ขอ resource) และ server (ให้ resource) ได้พร้อมกัน โดยไม่พึ่ง central server

Central server controls everythingCentral server ควบคุมทุกอย่าง
Clients receive service from serverClients รับบริการจาก server

Great for web apps, APIsเหมาะกับ web apps, APIs
🕸️ Peer-to-Peer
No central authorityไม่มี central authority

🕸️
Pure P2P

Hybrid P2P

Core ConceptsConcepts หลัก

🔑
DHT (Distributed Hash Table)
A distributed map of key→value across all nodes — each node is responsible for its own key range.แผนที่กระจาย key→value ทั่วทุก node — แต่ละ node รับผิดชอบ key range ของตัวเอง
  • Lookup: O(log N) hops
  • No central index
  • Self-organizing
🏠
Node ID & Routing
Each peer has a unique ID (hash) that determines which data it is responsible for storing.แต่ละ peer มี unique ID (hash) ใช้กำหนดว่า data ชิ้นไหนควรเก็บที่ peer ไหน
  • SHA1/SHA256 hash
  • Consistent hashing
  • Finger table (Chord)
🤝
Peer Discovery
How to find peers in the network — bootstrap nodes, mDNS, DHT crawl, tracker.วิธีค้นหา peers ในเครือข่าย — bootstrap nodes, mDNS, DHT crawl, tracker
  • Bootstrap nodes
  • mDNS (local network)
  • Peer exchange (PEX)
📦
Data Replication
Stores copies of data across multiple nodes for availability when a node leaves the network.เก็บ copy ของ data ไว้หลาย nodes เพื่อ availability เมื่อ node ออกจากเครือข่าย
  • Replication factor k
  • Auto-repair on churn
  • Merkle tree verification
Network ChallengesNetwork Challenges
🔄
Churn
Nodes join and leave constantly — the system must adapt and re-route data automatically.Nodes เข้าและออกตลอดเวลา — ระบบต้องปรับตัวและ re-route data อัตโนมัติ
🔓
NAT Traversal
Peers sit behind NAT routers — requires STUN/TURN/hole-punching to establish connections.Peers อยู่หลัง NAT router — ต้องใช้ STUN/TURN/hole-punching เพื่อ connect กัน

Attacker creates many fake nodes — requires Proof of Work or a reputation system to defend.ผู้โจมตีสร้าง fake nodes จำนวนมาก — ต้องการ Proof of Work หรือ reputation system
⚡
Eventual Consistency
Data updates propagate asynchronously — nodes may temporarily see different versions of the data.Data update กระจายไม่พร้อมกัน — nodes อาจเห็น data version ต่างกันชั่วคราว

peer/peer.go — Basic Peer Node
package peer

import (
	"crypto/sha1"
	"encoding/hex"
	"fmt"
	"net"
	"sync"
)

// PeerID — unique identifier for a node (hash-based)
type PeerID string

func NewPeerID(addr string) PeerID {
	h := sha1.Sum([]byte(addr))
	return PeerID(hex.EncodeToString(h[:]))
}

// PeerInfo — information about a peer in the network
type PeerInfo struct {
	ID   PeerID
	Addr string // "host:port"
}

// Node — a peer that is both client and server
type Node struct {
	mu       sync.RWMutex
	self     PeerInfo
	peers    map[PeerID]*PeerInfo
	data     map[string][]byte // local data store
	listener net.Listener
}

func NewNode(addr string) *Node {
	return &Node{
		self:  PeerInfo{ID: NewPeerID(addr), Addr: addr},
		peers: make(map[PeerID]*PeerInfo),
		data:  make(map[string][]byte),
	}
}

// Start — begins listening for incoming connections (acting as server)
func (n *Node) Start() error {
	ln, err := net.Listen("tcp", n.self.Addr)
	if err != nil {
		return fmt.Errorf("listen on %s: %w", n.self.Addr, err)
	}
	n.listener = ln
	go n.acceptLoop()
	fmt.Printf("[%s] Node started at %s\n", n.self.ID[:8], n.self.Addr)
	return nil
}

func (n *Node) acceptLoop() {
	for {
		conn, err := n.listener.Accept()
		if err != nil {
			return
		}
		go n.handleConnection(conn)
	}
}

func (n *Node) handleConnection(conn net.Conn) {
	defer conn.Close()
	// Protocol handling: read message type, dispatch
	buf := make([]byte, 4096)
	nRead, err := conn.Read(buf)
	if err != nil {
		return
	}
	msg := ParseMessage(buf[:nRead])
	n.handleMessage(conn, msg)
}
import * as crypto from 'crypto';
import * as WebSocket from 'ws';

// PeerID — unique identifier for a node (hash-based)
type PeerID = string;

function newPeerID(addr: string): PeerID {
  return crypto.createHash('sha1').update(addr).digest('hex');
}

// PeerInfo — information about a peer in the network
interface PeerInfo {
  id: PeerID;
  addr: string; // "host:port" or "ws://host:port"
}

// Node — a peer that is both client and server
class Node {
  private self: PeerInfo;
  private peers: Map = new Map();
  private data: Map = new Map();
  private server: WebSocket.Server | null = null;

  constructor(addr: string) {
    this.self = { id: newPeerID(addr), addr };
  }

  // start — begins listening for incoming WebSocket connections (acting as server)
  start(): void {
    const port = parseInt(this.self.addr.replace(':', ''), 10);
    this.server = new WebSocket.Server({ port });

    this.server.on('connection', (ws) => {
      this.handleConnection(ws);
    });

    console.log(`[${this.self.id.slice(0, 8)}] Node started at ${this.self.addr}`);
  }

  private handleConnection(ws: WebSocket): void {
    ws.on('message', (data: Buffer) => {
      const msg: Message = JSON.parse(data.toString());
      this.handleMessage(ws, msg);
    });
  }
}
peer/message.go — P2P Protocol Messages
package peer

import (
	"encoding/json"
	"fmt"
)

type MessageType string

const (
	MsgPing      MessageType = "PING"
	MsgPong      MessageType = "PONG"
	MsgStore     MessageType = "STORE"     // store key-value on this peer
	MsgFindValue MessageType = "FIND_VALUE" // find value by key
	MsgFindNode  MessageType = "FIND_NODE"  // find peers close to a key
	MsgPeerList  MessageType = "PEER_LIST"  // response with peer list
)

type Message struct {
	Type    MessageType     `json:"type"`
	From    PeerID          `json:"from"`
	Payload json.RawMessage `json:"payload"`
}

type StorePayload struct {
	Key   string `json:"key"`
	Value []byte `json:"value"`
}

type FindValuePayload struct {
	Key string `json:"key"`
}

type FindValueResponse struct {
	Found bool     `json:"found"`
	Value []byte   `json:"value,omitempty"`
	Peers []PeerInfo `json:"peers,omitempty"` // closer peers if not found
}

func ParseMessage(data []byte) Message {
	var msg Message
	json.Unmarshal(data, &msg)
	return msg
}

func (n *Node) handleMessage(conn interface{ Write([]byte) (int, error) }, msg Message) {
	switch msg.Type {
	case MsgPing:
		resp, _ := json.Marshal(Message{Type: MsgPong, From: n.self.ID})
		conn.Write(resp)

	case MsgStore:
		var p StorePayload
		json.Unmarshal(msg.Payload, &p)
		n.mu.Lock()
		n.data[p.Key] = p.Value
		n.mu.Unlock()
		fmt.Printf("[%s] Stored key=%s\n", n.self.ID[:8], p.Key)

	case MsgFindValue:
		var p FindValuePayload
		json.Unmarshal(msg.Payload, &p)
		n.mu.RLock()
		val, found := n.data[p.Key]
		n.mu.RUnlock()

		resp := FindValueResponse{Found: found, Value: val}
		if !found {
			// Return k-closest peers we know about
			resp.Peers = n.closestPeers(p.Key, 3)
		}
		payload, _ := json.Marshal(resp)
		out, _ := json.Marshal(Message{Type: MsgFindValue, From: n.self.ID, Payload: payload})
		conn.Write(out)
	}
}
peer/routing.go — Simple DHT-like Routing
package peer

import (
	"crypto/sha1"
	"encoding/binary"
	"net"
	"sort"
)

// distance XOR metric — lower = closer (Kademlia style)
func distance(a, b PeerID) uint32 {
	ha := sha1.Sum([]byte(a))
	hb := sha1.Sum([]byte(b))
	var d uint32
	for i := 0; i < 4; i++ {
		d ^= binary.BigEndian.Uint32(ha[i*4 : (i+1)*4])
		d ^= binary.BigEndian.Uint32(hb[i*4 : (i+1)*4])
	}
	return d
}

// closestPeers returns k peers with shortest XOR distance to key
func (n *Node) closestPeers(key string, k int) []PeerInfo {
	target := PeerID(key)
	n.mu.RLock()
	defer n.mu.RUnlock()

	type scored struct {
		peer PeerInfo
		dist uint32
	}
	var candidates []scored
	for _, p := range n.peers {
		candidates = append(candidates, scored{*p, distance(target, p.ID)})
	}
	sort.Slice(candidates, func(i, j int) bool {
		return candidates[i].dist < candidates[j].dist
	})

	result := make([]PeerInfo, 0, k)
	for i := 0; i < k && i < len(candidates); i++ {
		result = append(result, candidates[i].peer)
	}
	return result
}

// Connect — acting as client, connecting to a known peer
func (n *Node) Connect(addr string) error {
	conn, err := net.Dial("tcp", addr)
	if err != nil {
		return err
	}
	defer conn.Close()

	// Send ping
	msg, _ := json.Marshal(Message{Type: MsgPing, From: n.self.ID})
	conn.Write(msg)

	// Read pong, add peer to table
	buf := make([]byte, 1024)
	nRead, _ := conn.Read(buf)
	var resp Message
	json.Unmarshal(buf[:nRead], &resp)

	n.mu.Lock()
	n.peers[resp.From] = &PeerInfo{ID: resp.From, Addr: addr}
	n.mu.Unlock()
	return nil
}
peer/gossip.go — Gossip Protocol for peer discovery
package peer

import (
	"encoding/json"
	"fmt"
	"math/rand"
	"net"
	"time"
)

// Gossip periodically shares peer list with random neighbors
// — แพร่กระจายข้อมูลแบบ epidemic ทุก peer จะรู้จัก peer อื่นๆ เร็วขึ้น
func (n *Node) StartGossip(interval time.Duration) {
	ticker := time.NewTicker(interval)
	go func() {
		for range ticker.C {
			n.gossipOnce()
		}
	}()
}

func (n *Node) gossipOnce() {
	n.mu.RLock()
	peerList := make([]PeerInfo, 0, len(n.peers))
	for _, p := range n.peers {
		peerList = append(peerList, *p)
	}
	n.mu.RUnlock()

	if len(peerList) == 0 {
		return
	}

	// Pick a random peer to gossip with
	target := peerList[rand.Intn(len(peerList))]

	// Share our peer list with them
	payload, _ := json.Marshal(peerList)
	msg, _ := json.Marshal(Message{
		Type:    MsgPeerList,
		From:    n.self.ID,
		Payload: payload,
	})

	conn, err := net.DialTimeout("tcp", target.Addr, 2*time.Second)
	if err != nil {
		fmt.Printf("[gossip] cannot reach %s: %v\n", target.Addr, err)
		return
	}
	defer conn.Close()
	conn.Write(msg)
}
cmd/main.go — Start two peers and connect them
package main

import (
	"fmt"
	"time"

	"myapp/peer"
)

func main() {
	// Create Peer A
	nodeA := peer.NewNode(":4001")
	if err := nodeA.Start(); err != nil {
		panic(err)
	}

	// Create Peer B
	nodeB := peer.NewNode(":4002")
	if err := nodeB.Start(); err != nil {
		panic(err)
	}

	// B connects to A (bootstrap)
	time.Sleep(100 * time.Millisecond)
	if err := nodeB.Connect(":4001"); err != nil {
		fmt.Println("connect failed:", err)
		return
	}
	fmt.Println("Peer B connected to Peer A")

	// B stores data (will be routed to correct node)
	nodeB.StoreValue("user:123", []byte(`{"name":"Alice"}`))

	// A retrieves data (via DHT routing)
	val, found := nodeA.FindValue("user:123")
	if found {
		fmt.Printf("Found: %s\n", val)
	}

	// Start gossip so peers discover each other automatically
	nodeA.StartGossip(5 * time.Second)
	nodeB.StartGossip(5 * time.Second)

	// Keep running
	select {}
}
